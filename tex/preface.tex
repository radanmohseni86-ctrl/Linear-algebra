\chapter*{درباره‌ی این جزوه}
\addcontentsline{toc}{chapter}{درباره‌ی این جزوه}
\markboth{درباره‌ی این جزوه}{درباره‌ی این جزوه}

این جزوه بر پایه‌ی اسلایدهای درس \textbf{جبر خطی} (\en{CE282}) در دانشکده‌ی مهندسی کامپیوتر دانشگاه صنعتی شریف نوشته شده است. هدف این است که مطالب اسلایدها به شکل یک متن پیوسته، روان و کامل درآید؛ متنی که بشود آن را مثل یک کتاب از اول تا آخر خواند و چیزی جا نماند.

\section*{این جزوه چه چیزی به اسلایدها اضافه می‌کند؟}
\begin{itemize}
  \item \textbf{اثبات کامل قضیه‌ها:} هر جا در اسلایدها جای اثبات خالی مانده بود (\en{Proof:}) یا فقط ایده‌ی اثبات آمده بود، اثبات کامل و قدم‌به‌قدم نوشته شده است.
  \item \textbf{حل کامل مثال‌ها:} مثال‌هایی که در اسلایدها بدون پاسخ بودند، با جزئیات حل شده‌اند.
  \item \textbf{شهود و توضیح ساده:} قبل از تعریف‌های رسمی سعی شده تصویر ذهنی و انگیزه‌ی هر مفهوم گفته شود تا برای خواندن جزوه پیش‌نیاز خاصی لازم نباشد.
  \item \textbf{فصل صفر (پیش‌درآمد):} نمادها و ابزارهای پایه‌ای مثل مجموعه، سیگما، اعداد مختلط و میدان را مرور می‌کند. اگر با این‌ها آشنا هستید، می‌توانید از آن بگذرید.
  \item \textbf{اصلاح چند خطای کوچک:} در چند جا عدد یا گزاره‌ای در اسلایدها نادرست بود؛ این موارد درست شده و در پانویس توضیح داده شده است.
  \item \textbf{تمرین با پاسخ کوتاه} در پایان هر فصل و \textbf{واژه‌نامه‌ی انگلیسی به فارسی} در پیوست.
\end{itemize}

\section*{راهنمای جعبه‌ها}
مثل کتاب‌های درسی، هر نوع مطلب در یک جعبه‌ی رنگی جدا آمده است تا هنگام مرور، سریع پیدا شود:
\begingroup
\renewcommand{\theboxnum}{۲-۵}

\begin{definition}[نام مفهوم]
تعریف رسمی یک مفهوم جدید. واژه‌ی انگلیسی هم کنار هر اصطلاح آمده است.
\end{definition}
\begin{theorem}[نام قضیه]
یک حکم مهم که باید ثابت شود. معمولاً بلافاصله پس از آن، جعبه‌ی خاکستریِ «اثبات» می‌آید.
\end{theorem}
\begin{proof}
اثبات‌ها در این جعبه‌ی خاکستری نوشته می‌شوند و پایانشان با مربع توپر مشخص است.
\end{proof}
\begin{lemma}
لم‌ها (قضیه‌های کمکی) و نتیجه‌ها (حکم‌هایی که سریع از یک قضیه به دست می‌آیند) سبز تیره هستند.
\end{lemma}
\begin{example}
مثال‌ها نارنجی هستند و پاسخشان در جعبه‌ی «پاسخ» زیرشان می‌آید.
\end{example}
\begin{solution}
پاسخ مثال‌ها اینجا نوشته می‌شود.
\end{solution}
\begin{intuition}
توضیح شهودی یا هندسی؛ برای این‌که «حس» مفهوم را بگیریم.
\end{intuition}
\begin{note}
نکته‌های تکمیلی که خوب است در ذهن بمانند.
\end{note}
\begin{warning}
اشتباه‌های رایج و جاهایی که باید دقت کرد.
\end{warning}
\begin{summary}
خلاصه‌ی مهم‌ترین مطالب در پایان هر فصل.
\end{summary}
\endgroup

\section*{قراردادهای نمادگذاری}
\begin{itemize}
  \item بردارها را با حروف \textbf{پررنگ} می‌نویسیم: $\vx,\vy,\vb,\vv$. بردار صفر را با $\zero$ نشان می‌دهیم.
  \item ماتریس‌ها با حروف بزرگ نوشته می‌شوند: $A,B,E,L,U$. درایه‌ی سطر $i$ و ستون $j$ ماتریس $A$ را $A_{ij}$ یا $a_{ij}$ می‌نویسیم.
  \item اسکالرها (عددها) را با حروف کوچک یا یونانی نشان می‌دهیم: $c,\alpha,\beta,\lambda$.
  \item ترانهاده‌ی ماتریس $A$ را با $A\T$ نشان می‌دهیم. بردارها به‌طور پیش‌فرض \textbf{ستونی} هستند و $\vx\T$ یک بردار سطری است.
  \item اعداد درون متن فارسی با رقم فارسی و اعداد درون فرمول‌ها با رقم لاتین نوشته شده‌اند.
  \item شماره‌گذاری جعبه‌ها به صورت «شماره‌ی فصل - شماره‌ی جعبه» است؛ مثلاً «قضیه‌ی ۲-۵» یعنی جعبه‌ی پنجم از فصل ۲.
\end{itemize}
