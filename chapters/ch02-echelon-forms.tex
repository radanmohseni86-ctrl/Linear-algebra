\chapter{فرم‌های پلکانی و حل دستگاه‌های خطی}
\label{ch:echelon}

عملیات سطری جواب‌های دستگاه را تغییر نمی‌دهند. حالا سؤال این است: \emph{ماتریس را به چه شکلی برسانیم تا جواب‌ها خودشان را نشان دهند؟} پاسخ، فرم‌های پلکانی است. در این فصل حذف گاوسی، شرط وجود و یکتایی جواب، تجزیه‌ی $LU$ و تعبیر هندسی جواب‌ها را می‌بینیم.

\section{ماتریس سطری‌-تحویل‌یافته}

\begin{definition}[ماتریس سطری‌-تحویل‌یافته]
ماتریس $R$ \term{سطری‌-تحویل‌یافته}{Row-Reduced} است اگر:
\begin{enumerate}
  \item اولین درایه‌ی ناصفر هر سطر ناصفر (\term{درایه‌ی پیشرو}{Leading Entry}) برابر ۱ باشد؛
  \item هر ستونی که درایه‌ی پیشروی یک سطر را دارد، بقیه‌ی درایه‌هایش صفر باشد.
\end{enumerate}
\end{definition}

\begin{example}
کدام سطری‌-تحویل‌یافته‌اند؟
\[
A=I_n,\quad
B=\begin{bmatrix}1&0&0\\0&2&-1\\0&0&0\end{bmatrix},\quad
C=\begin{bmatrix}1&0&-3\\0&1&0\\0&0&1\end{bmatrix},\quad
D=\begin{bmatrix}1&0&0\\0&0&0\\0&0&-6\end{bmatrix}.
\]
\end{example}
\begin{solution}
\textbf{$A$ بله.} \textbf{$B$ خیر:} درایه‌ی پیشروی سطر دوم ۲ است. \textbf{$C$ خیر:} ستون سوم ۱ِ پیشروی سطر سوم را دارد اما در سطر اول $-3$ دارد. \textbf{$D$ خیر:} درایه‌ی پیشروی سطر سوم $-6$ است.
\end{solution}

\begin{theorem}\label{thm:row-reduced-exists}
هر ماتریس $m\times n$ با یک ماتریس سطری‌-تحویل‌یافته هم‌ارز سطری است.
\end{theorem}
\textbf{ایده:} سطرها را یکی‌یکی پردازش می‌کنیم: درایه‌ی پیشروی سطر را با تقسیم ۱ می‌کنیم و با جایگزینی، بقیه‌ی درایه‌های ستونش را صفر می‌کنیم. این کار درایه‌های پیشروی سطرهای قبلی را خراب نمی‌کند، چون سطر جدید در ستون‌های پیشروِ قبلی صفر است.

\section{فرم پلکانی و \lr{RREF}}

\begin{definition}[فرم پلکانی]
ماتریس در \term{فرم پلکانی}{Echelon Form} است اگر:
(۱) سطرهای ناصفر بالای سطرهای تمام‌صفر باشند؛
(۲) درایه‌ی پیشروی هر سطر سمت راست درایه‌ی پیشروی سطر بالایی باشد؛
(۳) زیر هر درایه‌ی پیشرو همه صفر باشد.
\end{definition}

\begin{definition}[\lr{RREF}]
ماتریسی در فرم پلکانی که سطری‌-تحویل‌یافته هم باشد (هر درایه‌ی پیشرو ۱ و تنها درایه‌ی ناصفر ستونش)، در \term{فرم پلکانی سطری‌-تحویل‌یافته}{Reduced Row Echelon Form, RREF} است. مثلاً $\begin{bmatrix}1&0&0&29\\0&1&0&16\\0&0&1&3\end{bmatrix}$؛ اگر این ماتریس افزوده باشد، جواب مستقیم خوانده می‌شود: $x_1=29,\ x_2=16,\ x_3=3$.
\end{definition}

\begin{theorem}
هر ماتریس با یک ماتریس RREF هم‌ارز سطری است.
\end{theorem}
\textbf{اثبات} مثل قضیه‌ی \ref{thm:row-reduced-exists} است؛ در پایان با جابه‌جایی سطرها، سطرهای صفر را پایین می‌بریم و سطرهای ناصفر را به ترتیب جای درایه‌ی پیشرو مرتب می‌کنیم.

\begin{example}
کدام در فرم RREF هستند؟
\[
A=0_{m\times n},\quad
B=\begin{bmatrix}1&0&0\\0&2&-1\\0&0&0\end{bmatrix},\quad
C=\begin{bmatrix}0&1\\1&0\end{bmatrix},\quad
D=\begin{bmatrix}0&1&-3&0&0.5\\0&0&0&1&2\\0&0&0&0&0\end{bmatrix}.
\]
\end{example}
\begin{solution}
\textbf{$A$ بله} (سطر ناصفری ندارد). \textbf{$B$ خیر} (درایه‌ی پیشروی ۲). \textbf{$C$ خیر:} درایه‌ی پیشروی سطر دوم سمت چپ سطر اول است. \textbf{$D$ بله.}
\end{solution}

\begin{lemma}[عملیات سطری روابط بین ستون‌ها را حفظ می‌کنند]\label{lem:col-relations}
اگر $A$ و $B$ هم‌ارز سطری باشند، برای هر $\vx$: \ $A\vx=\zero\iff B\vx=\zero$.
\end{lemma}
\begin{proof}
$B=PA$ با $P$ وارون‌پذیر. اگر $A\vx=\zero$ آن‌گاه $B\vx=P\zero=\zero$؛ و اگر $B\vx=\zero$ آن‌گاه $A\vx=P^{-1}B\vx=\zero$.
\end{proof}

\begin{note}
به کمک این لم می‌شود نشان داد \textbf{RREF هر ماتریس یکتاست}؛ پس از «$\RREF(A)$» حرف می‌زنیم. جای درایه‌های پیشرو را \term{مکان محوری}{Pivot Position} و ستونشان را \term{ستون محوری}{Pivot Column} می‌نامیم. در RREF: \hl{تعداد سطرهای ناصفر = تعداد ستون‌های محوری.}
\end{note}

\paragraph{الگوریتم کاهش سطری.} \textbf{رفت:} چپ‌ترین ستون ناصفر را پیدا کن، یک درایه‌ی ناصفر را (با جابه‌جایی) بالای آن بیاور و زیرش را صفر کن؛ همین کار را روی زیرماتریس باقی‌مانده تکرار کن (نتیجه: فرم پلکانی). \textbf{برگشت:} از راست‌ترین محور، بالای هر محور را صفر و محور را ۱ کن (نتیجه: RREF).

\begin{example}[کاهش سطری کامل]
ماتریس $A=\begin{bmatrix}0&3&-6&6&4&-5\\3&-7&8&-5&8&9\\3&-9&12&-9&6&15\end{bmatrix}$ را به RREF برسانید.
\end{example}
\begin{solution}
\[
\xrightarrow{R_1\leftrightarrow R_3}
\begin{bmatrix}3&-9&12&-9&6&15\\3&-7&8&-5&8&9\\0&3&-6&6&4&-5\end{bmatrix}
\xrightarrow{R_2\leftarrow R_2-R_1}
\begin{bmatrix}3&-9&12&-9&6&15\\0&2&-4&4&2&-6\\0&3&-6&6&4&-5\end{bmatrix}
\]
\[
\xrightarrow{R_3\leftarrow R_3-\frac32R_2}
\begin{bmatrix}3&-9&12&-9&6&15\\0&2&-4&4&2&-6\\0&0&0&0&1&4\end{bmatrix}.
\]
این فرم پلکانی است (ستون‌های محوری ۱، ۲ و ۵). با مرحله‌ی برگشت ($R_2\leftarrow R_2-2R_3$، $R_1\leftarrow R_1-6R_3$، $R_2\leftarrow\frac12R_2$، $R_1\leftarrow R_1+9R_2$، $R_1\leftarrow\frac13R_1$):
\[
\RREF(A)=\begin{bmatrix}1&0&-2&3&0&-24\\0&1&-2&2&0&-7\\0&0&0&0&1&4\end{bmatrix}.
\]
\end{solution}

\section{جواب‌های یک دستگاه خطی}

متغیرهای مربوط به ستون‌های محوری \term{متغیرهای پایه‌ای}{Basic Variables} و بقیه \term{متغیرهای آزاد}{Free Variables} هستند. مثلاً
\[
\left[\begin{array}{ccc|c}1&0&-5&1\\0&1&1&4\\0&0&0&0\end{array}\right]
\ \Longrightarrow\
x=1+5z,\quad y=4-z,\quad z\ \text{\fa{آزاد}} .
\]

\begin{theorem}[شرط سازگاری]\label{thm:consistency}
یک دستگاه خطی سازگار است اگر و تنها اگر آخرین ستون ماتریس افزوده محوری \emph{نباشد}؛ یعنی در فرم پلکانی آن سطری به شکل $\begin{bmatrix}0&\cdots&0&|&b\end{bmatrix}$ با $b\neq0$ وجود نداشته باشد.
\end{theorem}
\begin{proof}
طبق قضیه‌ی \ref{thm:row-equiv-solutions} کافی است دستگاه کاهش‌یافته را بررسی کنیم.

\textbf{(رفت)} اگر آخرین ستون محوری باشد، سطری به شکل $\begin{bmatrix}0&\cdots&0&|&b\end{bmatrix}$ با $b\neq0$ داریم، یعنی معادله‌ی $0x_1+\dots+0x_n=b$ که هیچ جوابی ندارد. پس دستگاه سازگار نیست.

\textbf{(برگشت)} اگر آخرین ستون محوری نباشد، در RREF هر سطر ناصفر یک متغیر پایه‌ای دارد که فقط در همان معادله آمده است. به متغیرهای آزاد مقدار دلخواه (مثلاً صفر) بدهید و هر متغیر پایه‌ای را از معادله‌ی خودش حساب کنید؛ این یک جواب است.
\end{proof}

\begin{theorem}[یکتایی]
اگر دستگاه سازگار باشد: بدون متغیر آزاد \textbf{دقیقاً یک جواب} دارد و با دست‌کم یک متغیر آزاد \textbf{بی‌شمار جواب}.
\end{theorem}
(متغیرهای پایه‌ای به‌طور یکتا از روی متغیرهای آزاد تعیین می‌شوند و هر متغیر آزاد هر مقداری می‌تواند بگیرد.)

\paragraph{یافتن همه‌ی جواب‌ها.} ماتریس افزوده را بنویسید، به فرم پلکانی برسانید و سازگاری را بررسی کنید؛ اگر سازگار است، تا RREF ادامه دهید و هر متغیر پایه‌ای را بر حسب متغیرهای آزاد بنویسید. مثلاً اگر $A$ در مثال قبل ماتریس افزوده‌ی یک دستگاه ۵ مجهولی باشد: $x_1=-24+2x_3-3x_4$، $x_2=-7+2x_3-2x_4$، $x_5=4$ و $x_3,x_4$ آزادند.

\begin{example}[سازگاری برای همه‌ی $\vb$ها]
آیا $A\vx=\vb$ با $A=\begin{bmatrix}1&3&4\\-4&2&-6\\-3&-2&-7\end{bmatrix}$ برای همه‌ی $\vb=(b_1,b_2,b_3)$ سازگار است؟
\end{example}
\begin{solution}
\[
\left[\begin{array}{ccc|c}1&3&4&b_1\\-4&2&-6&b_2\\-3&-2&-7&b_3\end{array}\right]
\sim
\left[\begin{array}{ccc|c}1&3&4&b_1\\0&14&10&b_2+4b_1\\0&0&0&b_1-\frac12b_2+b_3\end{array}\right].
\]
دستگاه فقط وقتی سازگار است که $b_1-\frac12b_2+b_3=0$؛ پس \textbf{برای همه‌ی $\vb$ها سازگار نیست} (مثلاً $\vb=(1,0,0)$). مشکل این است که $A$ در سطر سوم محور ندارد.
\end{solution}

\begin{note}
در حالت کلی: \hl{$A\vx=\vb$ برای هر $\vb$ جواب دارد اگر و تنها اگر $A$ در هر سطر یک مکان محوری داشته باشد.}
\end{note}

\begin{example}[درست یا نادرست؟]
«اگر $\aug{A}{\vb}$ در فرم RREF در هر سطر محور داشته باشد، $A\vx=\vb$ سازگار است.»
\end{example}
\begin{solution}
\textbf{نادرست.} محور ممکن است در ستون آخر باشد: در $\left[\begin{array}{cc|c}1&0&0\\0&0&1\end{array}\right]$ سطر دوم یعنی $0=1$. جمله‌ی درست درباره‌ی محور در هر سطرِ \emph{ماتریس ضرایب} $A$ است.
\end{solution}

\subsection{دستگاه‌های همگن}
دستگاه $A\vx=\zero$ همیشه \term{جواب بدیهی}{Trivial Solution} $\vx=\zero$ را دارد. هر جواب ناصفر را \term{نابدیهی}{Nontrivial} می‌نامیم.

\begin{theorem}\label{thm:homog-free}
$A\vx=\zero$ جواب نابدیهی دارد اگر و تنها اگر دست‌کم یک متغیر آزاد داشته باشد.
\end{theorem}
\begin{proof}
$A$ را به RREF برسانید. اگر همه‌ی متغیرها پایه‌ای باشند، معادله‌ی هر سطر می‌گوید آن متغیر صفر است؛ پس تنها جواب $\vx=\zero$ است. اگر $x_j$ آزاد باشد، با $x_j=t\neq0$ و محاسبه‌ی متغیرهای پایه‌ای، یک جواب ناصفر به دست می‌آید.
\end{proof}

مثلاً $\begin{bmatrix}1&3&4\\2&-1&2\\1&0&1\end{bmatrix}$ به $I$ کاهش می‌یابد و فقط جواب بدیهی دارد.

\begin{corollary}[مجهول‌های بیشتر از معادله‌ها]\label{cor:more-unknowns}
اگر دستگاه همگن $n$ مجهول و $m$ معادله داشته باشد و $n>m$، جواب نابدیهی دارد.
\end{corollary}
(حداکثر $m$ محور داریم، پس دست‌کم $n-m\ge1$ متغیر آزاد هست.)

\begin{theorem}\label{thm:solvable-combination}
$A\vx=\vb$ جواب دارد اگر و تنها اگر $\vb$ ترکیب خطی ستون‌های $A$ باشد.
\end{theorem}
\begin{proof}
$A\vx=x_1\va_1+\dots+x_n\va_n$. اگر $A\vx=\vb$، همین تساوی $\vb$ را ترکیب خطی ستون‌ها می‌نویسد. برعکس، اگر $\vb=c_1\va_1+\dots+c_n\va_n$، آن‌گاه $\vx=(c_1,\dots,c_n)$ جواب است.
\end{proof}

\begin{theorem}[جواب عمومی]\label{thm:general-solution}
اگر $\vp$ یک جواب $A\vx=\vb$ باشد، همه‌ی جواب‌ها به شکل $\vp+\vv_h$ هستند که $A\vv_h=\zero$.
\end{theorem}
(چون $A(\vx-\vp)=\vb-\vb=\zero$.) یعنی مجموعه‌ی جواب $A\vx=\vb$ همان جواب‌های $A\vx=\zero$ است که به اندازه‌ی $\vp$ جابه‌جا شده:

\section{تجزیه‌ی $LU$}

نوشتن یک ماتریس به صورت حاصل‌ضرب چند ماتریس دیگر را \term{تجزیه‌ی ماتریسی}{Matrix Factorization} می‌گوییم؛ مثل $30=2\times3\times5$ برای اعداد. تجزیه‌ی $LU$ از دل حذف گاوسی بیرون می‌آید و برای کامپیوتر بسیار مناسب است.

\begin{definition}[تجزیه‌ی $LU$]
برای ماتریس مربعی $A$، نمایش $A=LU$ که $L$ \term{پایین‌مثلثی واحد}{Unit Lower Triangular} (بالای قطر صفر و روی قطر ۱) و $U$ بالامثلثی است، \term{تجزیه‌ی $LU$}{LU Factorization} نام دارد.
\end{definition}

\begin{example}
تجزیه‌ی $LU$ ماتریس $A=\begin{bmatrix}2&1&3\\4&-1&3\\-2&5&5\end{bmatrix}$ را از روی حذف گاوسی بیابید.
\end{example}
\begin{solution}
\[
A\xrightarrow[R_3\leftarrow R_3+R_1]{R_2\leftarrow R_2-2R_1}
\begin{bmatrix}2&1&3\\0&-3&-3\\0&6&8\end{bmatrix}
\xrightarrow{R_3\leftarrow R_3+2R_2}
\begin{bmatrix}2&1&3\\0&-3&-3\\0&0&2\end{bmatrix}=U .
\]
با ماتریس‌های مقدماتی $E_1,E_2,E_3$ این سه عمل، $E_3E_2E_1A=U$؛ پس
\[
A=E_1^{-1}E_2^{-1}E_3^{-1}U=\underbrace{\begin{bmatrix}1&0&0\\2&1&0\\-1&-2&1\end{bmatrix}}_{L}U .
\]
\end{solution}

\begin{theorem}[وجود تجزیه‌ی $LU$]
اگر ماتریس مربعی $A$ را بتوان \textbf{بدون جابه‌جایی سطرها} به فرم پلکانی رساند، $A$ تجزیه‌ی $LU$ دارد.
\end{theorem}
\begin{proof}
بدون جابه‌جایی، حذف فقط از عمل‌های $R_i\leftarrow R_i-\ell R_j$ با $j<i$ استفاده می‌کند؛ ماتریس مقدماتی هر کدام (و وارونش) پایین‌مثلثی واحد است. پس $E_k\cdots E_1A=U$ و $A=E_1^{-1}\cdots E_k^{-1}U$. حاصل‌ضرب ماتریس‌های پایین‌مثلثی واحد هم پایین‌مثلثی واحد است (درایه‌ی $(i,j)$ با $i<j$ صفر و درایه‌ی قطری $1\cdot1$ است). پس $L=E_1^{-1}\cdots E_k^{-1}$.
\end{proof}

\begin{warning}
اگر جابه‌جایی لازم باشد، $L$ دیگر پایین‌مثلثی نیست. مثلاً $\begin{bmatrix}0&1\\1&0\end{bmatrix}$ تجزیه‌ی $LU$ ندارد.
\end{warning}

\begin{note}[راه سریع ساختن $L$]
لازم نیست ماتریس‌های مقدماتی را ضرب کنیم: برای هر عمل $R_i\leftarrow R_i-k\,R_j$، عدد $k$ در درایه‌ی $(i,j)$ ماتریس $L$ نوشته می‌شود و قطر $L$ تمام ۱ است.
\end{note}

\begin{example}
تجزیه‌ی $LU$ ماتریس $A=\begin{bmatrix}6&-2&0\\9&-1&1\\3&7&5\end{bmatrix}$ را بیابید.
\end{example}
\begin{solution}
\[
A\xrightarrow[R_3\leftarrow R_3-\frac36R_1]{R_2\leftarrow R_2-\frac96R_1}
\begin{bmatrix}6&-2&0\\0&2&1\\0&8&5\end{bmatrix}
\xrightarrow{R_3\leftarrow R_3-4R_2}
\begin{bmatrix}6&-2&0\\0&2&1\\0&0&1\end{bmatrix}=U,
\qquad
L=\begin{bmatrix}1&0&0\\ \frac32&1&0\\ \frac12&4&1\end{bmatrix}.
\]
\end{solution}

برای ماتریس غیرمربعی $m\times n$ هم می‌توان نوشت $A=LU$ که $L$ مربعی $m\times m$ و $U$ در \textbf{فرم پلکانی} است. مثلاً
\[
\begin{bmatrix}2&4&2&1\\1&3&1&1\\3&7&4&2\end{bmatrix}
=\begin{bmatrix}1&0&0\\ \frac12&1&0\\ \frac32&1&1\end{bmatrix}
\begin{bmatrix}2&4&2&1\\0&1&0&0.5\\0&0&1&0\end{bmatrix}
\]
(با عمل‌های $R_2-\frac12R_1$، $R_3-\frac32R_1$ و $R_3-R_2$).

\subsection{حل دستگاه با تجزیه‌ی $LU$}
\begin{keyformula}
\[
A\vx=\vb\iff L\underbrace{(U\vx)}_{\vy}=\vb\iff
\begin{cases}L\vy=\vb & \text{\fa{(از بالا به پایین)}}\\ U\vx=\vy & \text{\fa{(از پایین به بالا)}}\end{cases}
\]
\end{keyformula}

\begin{example}
با تجزیه‌ی $LU$ مثال قبل، $A\vx=\vb$ را برای $A=\begin{bmatrix}2&1&3\\4&-1&3\\-2&5&5\end{bmatrix}$ و $\vb=(1,-4,9)$ حل کنید.
\end{example}
\begin{solution}
\textbf{$L\vy=\vb$:} $y_1=1$، \ $y_2=-4-2y_1=-6$، \ $y_3=9+y_1+2y_2=-2$.

\textbf{$U\vx=\vy$:} $2x_3=-2\Rightarrow x_3=-1$؛ \ $-3x_2-3x_3=-6\Rightarrow x_2=3$؛ \ $2x_1+x_2+3x_3=1\Rightarrow x_1=\frac12$. پس $\vx=(\frac12,3,-1)$.
\end{solution}

\begin{note}[چرا $LU$؟]
خود تجزیه حدود $\frac23n^3$ عمل حسابی می‌خواهد، اما بعد از آن حل هر دستگاه با $\vb$ جدید فقط حدود $2n^2$ عمل لازم دارد.
\end{note}

\begin{intuition}
حل دستگاه $m$ معادله با ۳ مجهول یعنی پیدا کردن محل برخورد $m$ صفحه؛ جواب می‌تواند تهی، یک نقطه، یک خط یا یک صفحه باشد. متغیرهای آزاد همان پارامترهای شکل برداری ($s,t$) هستند.
\end{intuition}

\begin{note}[جابه‌جا کردن ستون‌ها]
جابه‌جا کردن ستون‌های ماتریس ضرایب یعنی تغییر نام متغیرها؛ مجاز است به شرط این‌که حساب کنیم هر ستون مال کدام متغیر است. ستون آخر ($\vb$) را نباید با ستون‌های ضرایب جابه‌جا کرد.
\end{note}

\begin{example}
جواب‌ها را به شکل برداری بنویسید:
(الف) $A=\begin{bmatrix}3&5&-4\\-3&-2&4\\6&1&-8\end{bmatrix}$ و $\vb=(7,-1,-4)$؛ \
(ب) $A\vx=\zero$ با $A=\begin{bmatrix}1&0&-8&-7\\0&1&4&3\\0&0&0&0\end{bmatrix}$؛ \
(ج) $\aug{A}{\vb}=\left[\begin{array}{cccc|c}1&2&-1&1&3\\1&1&-1&1&1\\1&3&-1&1&5\end{array}\right]$.
\end{example}
\begin{solution}
\textbf{(الف)} RREF برابر $\left[\begin{array}{ccc|c}1&0&-\frac43&-1\\0&1&0&2\\0&0&0&0\end{array}\right]$ است؛ پس $\vx=(-1,2,0)+t(\frac43,0,1)$: خطی که از $(-1,2,0)$ می‌گذرد.

\textbf{(ب)} $x_1=8x_3+7x_4$ و $x_2=-4x_3-3x_4$؛ پس $\vx=s(8,-4,1,0)+t(7,-3,0,1)$.

\textbf{(ج)} RREF برابر $\left[\begin{array}{cccc|c}1&0&-1&1&-1\\0&1&0&0&2\\0&0&0&0&0\end{array}\right]$ است؛ پس $\vx=(-1,2,0,0)+s(1,0,1,0)+t(-1,0,0,1)$.
\end{solution}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item هر ماتریس با یک و فقط یک ماتریس RREF هم‌ارز سطری است؛ عملیات سطری روابط بین ستون‌ها را حفظ می‌کنند.
  \item دستگاه سازگار است $\iff$ ستون آخر ماتریس افزوده محوری نیست. اگر سازگار باشد: بدون متغیر آزاد یک جواب، با متغیر آزاد بی‌شمار جواب.
  \item $A\vx=\zero$ جواب نابدیهی دارد $\iff$ متغیر آزاد دارد (مثلاً وقتی $n>m$).
  \item $A\vx=\vb$ جواب دارد $\iff$ $\vb$ ترکیب خطی ستون‌های $A$ است؛ جواب عمومی = یک جواب خصوصی + جواب‌های همگن.
  \item اگر حذف بدون جابه‌جایی ممکن باشد $A=LU$ و درایه‌های $L$ همان ضرب‌کننده‌های حذف‌اند.
\end{itemize}
\end{summary}

\subsection*{تمرین‌ها}
\begin{enumerate}
  \item RREF ماتریس $\begin{bmatrix}1&2&3\\4&5&6\\7&8&9\end{bmatrix}$ را بیابید.
  \item به ازای چه $h$هایی $\left[\begin{array}{cc|c}1&h&4\\3&6&8\end{array}\right]$ ماتریس افزوده‌ی یک دستگاه سازگار است؟
  \item تجزیه‌ی $LU$ ماتریس $\begin{bmatrix}2&5\\-4&-7\end{bmatrix}$ را بیابید.
  \item درست یا نادرست: «هر دستگاه همگن با ۳ معادله و ۴ مجهول جواب نابدیهی دارد.»
\end{enumerate}

  \item سطر دوم پس از حذف $\begin{bmatrix}0&6-3h&|&-4\end{bmatrix}$ است؛ پس برای $h\neq2$ سازگار است.
  \item $L=\begin{bmatrix}1&0\\-2&1\end{bmatrix}$ و $U=\begin{bmatrix}2&5\\0&3\end{bmatrix}$.
  \item درست (نتیجه‌ی \ref{cor:more-unknowns}).
\end{enumerate}
