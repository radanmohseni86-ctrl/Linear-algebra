\chapter{استقلال خطی}
\label{ch:independence}

وقتی یک زیرفضا را با چند بردار تولید می‌کنیم، ممکن است بعضی بردارها \emph{اضافه} باشند؛ یعنی بدون آن‌ها هم همان زیرفضا تولید شود، چون خودشان ترکیبی از بقیه‌اند. مفهوم \textbf{استقلال خطی} دقیقاً همین «اضافه نبودن» را اندازه می‌گیرد. این مفهوم پلی است به سمت \emph{پایه} و \emph{بُعد} در فصل بعد.

\section{انگیزه: مسئله‌ی قیمت خانه}

فرض کنید می‌خواهیم قیمت خانه ($y$) را از روی متراژ آن ($f$) پیش‌بینی کنیم. داده‌هایی به شکل $(f^{(1)},y^{(1)})$، $(f^{(2)},y^{(2)})$، …، $(f^{(N)},y^{(N)})$ داریم و ساده‌ترین مدل، یک خط است:
\[
y=af+c .
\]
\begin{center}
\begin{tikzpicture}[scale=0.75]
  \draw[->] (0,0)--(5.4,0) node[below] {\small \fa{متراژ}: $f$};
  \draw[->] (0,0)--(0,4.3) node[left] {\small \fa{قیمت}: $y$};
  \foreach \x/\y in {0.6/0.9, 1.1/1.4, 1.6/1.6, 2.0/2.3, 2.6/2.5, 3.1/3.0, 3.6/3.2, 4.2/3.9, 4.6/3.8} \fill[excolor] (\x,\y) circle (2.2pt);
  \draw[thick, primary] (0,0.45)--(5,4.2) node[right] {\small $y=af+c$};
  \node[excolor, font=\small] at (2.15,2.75) {$(f^{(i)},y^{(i)})$};
\end{tikzpicture}
\end{center}
هر داده یک معادله‌ی خطی بر حسب مجهول‌های $a$ و $c$ می‌دهد: $af^{(i)}+c=y^{(i)}$. پس همه‌ی داده‌ها با هم یک دستگاه $A\vx=\vb$ می‌سازند:
\[
\underbrace{\begin{bmatrix}f^{(1)}&1\\ f^{(2)}&1\\ \vdots&\vdots\\ f^{(N)}&1\end{bmatrix}}_{A}
\underbrace{\begin{bmatrix}a\\c\end{bmatrix}}_{\vx}
=\underbrace{\begin{bmatrix}y^{(1)}\\ y^{(2)}\\ \vdots\\ y^{(N)}\end{bmatrix}}_{\vb}.
\]
در واقعیت از ویژگی‌های بیشتری استفاده می‌کنیم: تعداد اتاق، متراژ، تعداد اتاق‌خواب، سن بنا، ویژگی‌های آدرس (خیابان، کوچه، …)، متراژ بخش ۱ تا ۴، تعداد طبقه، تعداد سرویس بهداشتی و … . هر ویژگی یک \textbf{ستون} ماتریس $A$ است. سؤال مهم: \emph{کدام ویژگی‌ها به بقیه وابسته‌اند؟}

مثلاً اگر «متراژ کل = متراژ بخش ۱ + بخش ۲ + بخش ۳ + بخش ۴» باشد، ستون متراژ کل دقیقاً ترکیب خطی چهار ستون دیگر است و هیچ اطلاع تازه‌ای به مدل نمی‌دهد؛ فقط حجم محاسبه را بیشتر و جواب را غیریکتا می‌کند. (اما «تعداد اتاق‌خواب» و «تعداد اتاق» فقط \emph{همبسته‌اند}، نه دقیقاً وابسته‌ی خطی.) زبان دقیق این بحث، استقلال خطی است.

\section{تعریف استقلال خطی}

\subsection{تعریف جبری}
\begin{definition}[وابستگی و استقلال خطی]
فرض کنید $\vv_1,\dots,\vv_n$ بردارهایی در فضای برداری $V$ باشند. معادله‌ی
\[
\lambda_1\vv_1+\lambda_2\vv_2+\dots+\lambda_n\vv_n=\zero
\]
را در نظر بگیرید. این معادله همیشه جواب بدیهی $\lambda_1=\dots=\lambda_n=0$ را دارد.
\begin{itemize}
  \item بردارها \term{وابسته‌ی خطی}{Linearly Dependent} هستند اگر این معادله جوابی داشته باشد که در آن \textbf{دست‌کم یک} $\lambda_i\neq0$. به چنین تساوی‌ای یک \textbf{رابطه‌ی وابستگی خطی} می‌گوییم.
  \item بردارها \term{مستقل خطی}{Linearly Independent} هستند اگر این معادله \textbf{فقط} جواب بدیهی داشته باشد؛ یعنی
  \[
  \lambda_1\vv_1+\dots+\lambda_n\vv_n=\zero\ \Longrightarrow\ \lambda_1=\dots=\lambda_n=0 .
  \]
\end{itemize}
\end{definition}

به زبان ساده (این را در قضیه‌ی \ref{thm:dep-char} ثابت می‌کنیم):
\begin{itemize}
  \item یک مجموعه \textbf{وابسته} است اگر دست‌کم یکی از بردارهایش را بتوان به صورت ترکیب خطی بقیه نوشت.
  \item یک مجموعه \textbf{مستقل} است اگر هیچ برداری در آن ترکیب خطی بقیه نباشد.
\end{itemize}

\begin{note}[زبان ماتریسی]
اگر $\vv_1,\dots,\vv_n\in\R^m$ را ستون‌های ماتریس $A=\begin{bmatrix}\vv_1&\cdots&\vv_n\end{bmatrix}$ بگیریم، معادله‌ی بالا همان $A\vx=\zero$ است. پس:
\begin{keyformula}[top=2mm,bottom=2mm]
\centering
ستون‌های $A$ مستقل خطی‌اند
\ $\iff$\ 
$A\vx=\zero$ فقط جواب بدیهی دارد
\ $\iff$\ 
$A\vx=\zero$ متغیر آزاد ندارد.
\end{keyformula}
(تساوی دوم از قضیه‌ی \ref{thm:homog-free} می‌آید.) این یعنی برای بررسی استقلال، کافی است ماتریس را کاهش سطری دهیم و ببینیم \textbf{همه‌ی ستون‌ها محوری هستند یا نه}.
\end{note}

\subsection{تعبیر هندسی}
\begin{intuition}
از نگاه هندسی، مجموعه‌ای از بردارها مستقل خطی است اگر \textbf{بُعد پوشش خطی آن‌ها با تعداد بردارها برابر باشد} (مفهوم بُعد را در فصل بعد دقیق می‌کنیم). مثلاً:
\begin{itemize}
  \item دو بردار وابسته‌اند اگر و تنها اگر \textbf{هم‌راستا} باشند (پوششان فقط یک خط است، نه یک صفحه).
  \item سه بردار در $\R^3$ وابسته‌اند اگر و تنها اگر \textbf{هم‌صفحه} باشند.
\end{itemize}
\end{intuition}

\begin{example}
آیا این مجموعه‌های بردار در $\R^2$ مستقل خطی‌اند؟
\begin{center}
\begin{tikzpicture}[scale=0.85,>=Latex]
  \draw[->, very thick] (0,0)--(-0.9,2.2);
  \draw[->, very thick, gray] (0,0)--(-0.45,1.1);
  \node at (-0.2,-0.4) {\lr{A)}};
  \begin{scope}[xshift=3cm]
  \draw[->, very thick] (0,0)--(-0.9,2.2);
  \draw[->, very thick, gray] (0,0)--(0.7,1.6);
  \node at (-0.1,-0.4) {\lr{B)}};
  \end{scope}
  \begin{scope}[xshift=6cm]
  \draw[->, very thick] (0,0)--(-0.9,2.2);
  \draw[->, very thick] (0,0)--(0.1,1.9);
  \draw[->, very thick, gray] (0,0)--(0.7,1.6);
  \node at (-0.1,-0.4) {\lr{C)}};
  \end{scope}
\end{tikzpicture}
\end{center}
\end{example}
\begin{solution}
\textbf{(A) وابسته:} دو بردار هم‌راستا هستند؛ یکی مضرب دیگری است و پوششان فقط یک خط (یک‌بعدی) است.
\textbf{(B) مستقل:} دو بردار غیرهم‌راستا؛ پوششان کل صفحه (دوبعدی) است.
\textbf{(C) وابسته:} سه بردار در $\R^2$؛ پوششان حداکثر دوبعدی است، پس نمی‌توانند مستقل باشند (قضیه‌ی \ref{thm:more-than-n}).
\end{solution}

\subsection{مثال‌های عددی}
\begin{example}
وابسته یا مستقل؟
\begin{enumerate}
  \item $\vv_1=\begin{bmatrix}1\\2\\3\end{bmatrix},\ \vv_2=\begin{bmatrix}4\\5\\6\end{bmatrix},\ \vv_3=\begin{bmatrix}2\\1\\0\end{bmatrix}$.
  \item $\vv_1=\begin{bmatrix}3\\1\end{bmatrix},\ \vv_2=\begin{bmatrix}6\\2\end{bmatrix}$.
  \item $\vv_1=\begin{bmatrix}3\\2\end{bmatrix},\ \vv_2=\begin{bmatrix}6\\2\end{bmatrix}$.
\end{enumerate}
\end{example}
\begin{solution}
\textbf{۱. وابسته.} با کمی دقت می‌بینیم
\[
-2\vv_1+\vv_2=\begin{bmatrix}-2\\-4\\-6\end{bmatrix}+\begin{bmatrix}4\\5\\6\end{bmatrix}=\begin{bmatrix}2\\1\\0\end{bmatrix}=\vv_3 ,
\]
یعنی $-2\vv_1+\vv_2-\vv_3=\zero$ یک رابطه‌ی وابستگی است. اگر این رابطه را حدس نزنیم، روش سیستماتیک کاهش سطری است:
\[
\begin{bmatrix}1&4&2\\2&5&1\\3&6&0\end{bmatrix}\sim\begin{bmatrix}1&0&-2\\0&1&1\\0&0&0\end{bmatrix}.
\]
ستون سوم محوری نیست، پس $x_3$ آزاد است و $A\vx=\zero$ جواب نابدیهی دارد؛ با $x_3=1$: $x_1=2,\ x_2=-1$، یعنی $2\vv_1-\vv_2+\vv_3=\zero$ (همان رابطه).

\textbf{۲. وابسته.} $\vv_2=2\vv_1$.

\textbf{۳. مستقل.} هیچ عددی مثل $\lambda$ نیست که $\vv_2=\lambda\vv_1$ (از مؤلفه‌ی اول $\lambda=2$ و از دومی $\lambda=1$). برای دو بردار، وابستگی یعنی یکی مضرب دیگری باشد (قضیه‌ی \ref{thm:dep-char})؛ پس مستقل‌اند.
\end{solution}

\section{ویژگی‌های استقلال خطی}

\begin{theorem}[بردار صفر]\label{thm:zero-dep}
هر مجموعه از بردارها که بردار صفر را در خود داشته باشد، حتماً وابسته‌ی خطی است.
\end{theorem}
\begin{proof}
فرض کنید مجموعه $\{\vv_1,\dots,\vv_k,\zero\}$ باشد. استقلال یعنی از
$c_1\vv_1+\dots+c_k\vv_k+c_{k+1}\zero=\zero$
نتیجه شود همه‌ی ضرایب صفرند. اما اگر بگذاریم $c_1=\dots=c_k=0$ و $c_{k+1}=1$، تساوی برقرار است چون $1\cdot\zero=\zero$. پس یک جواب نابدیهی داریم و مجموعه وابسته است.
\end{proof}

\begin{theorem}[مشخصه‌ی مجموعه‌های وابسته]\label{thm:dep-char}
مجموعه‌ی اندیس‌دار $S=\{\vv_1,\dots,\vv_n\}$ با دو بردار یا بیشتر، وابسته‌ی خطی است اگر و تنها اگر دست‌کم یکی از بردارهای $S$ ترکیب خطی بقیه باشد.

در واقع اگر $S$ وابسته باشد و $\vv_1\neq\zero$، آن‌گاه یک $\vv_j$ (با $j>1$) هست که ترکیب خطی \textbf{بردارهای قبل از خودش}، یعنی $\vv_1,\dots,\vv_{j-1}$، است.
\end{theorem}
\begin{proof}
\textbf{(برگشت)} اگر یک $\vv_j$ ترکیب خطی بقیه باشد، می‌توانیم $\vv_j$ را از دو طرف کم کنیم و یک رابطه‌ی وابستگی با ضریب ناصفر $-1$ برای $\vv_j$ بسازیم. مثلاً اگر $\vv_1=c_2\vv_2+c_3\vv_3$، آن‌گاه
\[
\zero=(-1)\vv_1+c_2\vv_2+c_3\vv_3+0\vv_4+\dots+0\vv_n .
\]
پس $S$ وابسته است.

\textbf{(رفت)} فرض کنید $S$ وابسته باشد. اگر $\vv_1=\zero$، آن‌گاه $\vv_1$ ترکیب خطی (بدیهیِ) بقیه است: $\vv_1=0\vv_2+\dots+0\vv_n$. پس فرض کنید $\vv_1\neq\zero$. ضرایبی مثل $c_1,\dots,c_n$ که همه صفر نیستند وجود دارند با
\[
c_1\vv_1+c_2\vv_2+\dots+c_n\vv_n=\zero .
\]
فرض کنید $j$ \textbf{بزرگ‌ترین} اندیسی باشد که $c_j\neq0$. اگر $j=1$ بود، رابطه می‌شد $c_1\vv_1=\zero$ با $c_1\neq0$، که طبق قضیه‌ی \ref{thm:vs-basic} نتیجه می‌دهد $\vv_1=\zero$؛ ناممکن است. پس $j>1$ و
\[
c_1\vv_1+\dots+c_j\vv_j+0\vv_{j+1}+\dots+0\vv_n=\zero
\ \Longrightarrow\
c_j\vv_j=-c_1\vv_1-\dots-c_{j-1}\vv_{j-1}
\]
\[
\Longrightarrow\ \vv_j=\Big(-\frac{c_1}{c_j}\Big)\vv_1+\dots+\Big(-\frac{c_{j-1}}{c_j}\Big)\vv_{j-1}.
\]
پس $\vv_j$ ترکیب خطی بردارهای قبل از خودش است (و در نتیجه ترکیب خطی بقیه).
\end{proof}

\begin{warning}
قضیه‌ی \ref{thm:dep-char} \textbf{نمی‌گوید} که در یک مجموعه‌ی وابسته \emph{هر} بردار ترکیب خطی بقیه است. مثلاً در $\{(1,0),(2,0),(0,1)\}$ بردار $(0,1)$ ترکیب خطی دو بردار دیگر نیست، با این‌که مجموعه وابسته است (چون $(2,0)=2(1,0)$).
\end{warning}

\begin{corollary}[دو بردار]
دو بردار $\vv_1,\vv_2$ وابسته‌اند اگر و تنها اگر یکی مضرب اسکالر دیگری باشد.
\end{corollary}

\begin{theorem}[تعداد زیاد، یعنی وابستگی]\label{thm:more-than-n}
\begin{itemize}
  \item هر مجموعه‌ی $p$تایی از بردارهای $\R^n$ با $p>n$ \textbf{حتماً وابسته} است.
  \item مجموعه‌ای با $p\le n$ بردار در $\R^n$ \emph{ممکن است} مستقل باشد (و ممکن است نباشد).
\end{itemize}
\end{theorem}
\begin{proof}
فرض کنید $A=\begin{bmatrix}\vv_1&\cdots&\vv_p\end{bmatrix}$ ماتریسی $n\times p$ باشد. معادله‌ی $A\vx=\zero$ یک دستگاه همگن با $n$ معادله و $p$ مجهول است. اگر $p>n$، طبق نتیجه‌ی \ref{cor:more-unknowns} جواب نابدیهی دارد؛ پس ستون‌ها وابسته‌اند.

برای بخش دوم: بردارهای استاندارد $\ve_1,\dots,\ve_p$ (با $p\le n$) مستقل‌اند، چون $c_1\ve_1+\dots+c_p\ve_p=(c_1,\dots,c_p,0,\dots,0)$ فقط وقتی صفر است که همه‌ی $c_i$ صفر باشند. از طرف دیگر $\{\vv,2\vv\}$ با $p=2\le n$ وابسته است.
\end{proof}

\begin{example}
وابسته یا مستقل؟ (سعی کنید بدون محاسبه‌ی زیاد جواب دهید.)
\[
\text{\fa{الف})}\ \begin{bmatrix}1\\7\\6\end{bmatrix},\begin{bmatrix}2\\0\\9\end{bmatrix},\begin{bmatrix}3\\1\\5\end{bmatrix},\begin{bmatrix}4\\1\\8\end{bmatrix}
\qquad
\text{\fa{ب})}\ \begin{bmatrix}2\\3\\5\end{bmatrix},\begin{bmatrix}0\\0\\0\end{bmatrix},\begin{bmatrix}1\\1\\8\end{bmatrix}
\qquad
\text{\fa{ج})}\ \begin{bmatrix}-2\\4\\6\\10\end{bmatrix},\begin{bmatrix}3\\-6\\-9\\15\end{bmatrix}
\]
\end{example}
\begin{solution}
\textbf{(الف) وابسته:} چهار بردار در $\R^3$؛ چون $p=4>3=n$، طبق قضیه‌ی \ref{thm:more-than-n}.
\textbf{(ب) وابسته:} بردار صفر در مجموعه است (قضیه‌ی \ref{thm:zero-dep}).
\textbf{(ج) مستقل:} دو بردارند و هیچ‌کدام مضرب دیگری نیست: اگر بردار دوم $\lambda$ برابر اولی بود، از مؤلفه‌ی اول $\lambda=-\frac32$ و از مؤلفه‌ی چهارم $\lambda=\frac32$ به دست می‌آمد.
\end{solution}

\subsection{حذف بردارهای اضافی}
\begin{theorem}\label{thm:remove-dependent}
فرض کنید $\vv_1,\dots,\vv_n$ وابسته‌ی خطی باشند و
\[
c_1\vv_1+c_2\vv_2+\dots+c_n\vv_n=\zero\qquad\text{\fa{با}}\ c_1\neq0 .
\]
آن‌گاه
$\Span\{\vv_1,\dots,\vv_n\}=\Span\{\vv_2,\dots,\vv_n\}$.
\end{theorem}
\begin{proof}
روشن است که $\Span\{\vv_2,\dots,\vv_n\}\subseteq\Span\{\vv_1,\dots,\vv_n\}$ (کافی است ضریب $\vv_1$ را صفر بگیریم). برعکس، چون $c_1\neq0$:
\[
\vv_1=-\frac{c_2}{c_1}\vv_2-\dots-\frac{c_n}{c_1}\vv_n .
\]
حالا هر ترکیب خطی $a_1\vv_1+a_2\vv_2+\dots+a_n\vv_n$ را با جایگذاری $\vv_1$ می‌شود فقط بر حسب $\vv_2,\dots,\vv_n$ نوشت:
\[
a_1\vv_1+\dots+a_n\vv_n=\Big(a_2-\frac{a_1c_2}{c_1}\Big)\vv_2+\dots+\Big(a_n-\frac{a_1c_n}{c_1}\Big)\vv_n .
\]
پس دو پوشش خطی برابرند.
\end{proof}

\begin{intuition}
وقتی یک فضا را به صورت پوشش خطی یک فهرست از بردارها می‌نویسیم، دوست داریم فهرست تا جای ممکن \textbf{کوتاه} باشد. قضیه‌ی بالا می‌گوید تا وقتی فهرست وابسته است، می‌توانیم یک بردار را (که ضریبش در رابطه‌ی وابستگی ناصفر است) حذف کنیم بی‌آن‌که پوشش عوض شود. با تکرار این کار به یک فهرست مستقل می‌رسیم که همان فضا را تولید می‌کند؛ این همان \emph{پایه} است (فصل ۶).
\end{intuition}

\subsection{یکتایی ضرایب}
\begin{theorem}[یکتایی نمایش]\label{thm:unique-coeffs}
فرض کنید $\vx$ ترکیب خطیِ بردارهای مستقل خطی $\vv_1,\dots,\vv_n$ باشد:
\[
\vx=\beta_1\vv_1+\dots+\beta_n\vv_n .
\]
آن‌گاه ضرایب $\beta_1,\dots,\beta_n$ \textbf{یکتا} هستند.
\end{theorem}
\begin{proof}
فرض کنید $\vx$ دو نمایش داشته باشد:
\[
\vx=\beta_1\vv_1+\dots+\beta_n\vv_n=\gamma_1\vv_1+\dots+\gamma_n\vv_n .
\]
با کم کردن دو نمایش از هم:
\[
(\beta_1-\gamma_1)\vv_1+\dots+(\beta_n-\gamma_n)\vv_n=\zero .
\]
چون بردارها مستقل خطی‌اند، این معادله فقط جواب بدیهی دارد:
$\beta_1-\gamma_1=\dots=\beta_n-\gamma_n=0$.
پس برای همه‌ی $i$ها $\beta_i=\gamma_i$ و ضرایب یکتا هستند.
\end{proof}

\begin{note}
عکس این قضیه هم درست است: اگر بردار $\zero$ فقط به یک شکل (یعنی با همه‌ی ضرایب صفر) نوشته شود، بردارها مستقل‌اند؛ این همان تعریف استقلال است. پس «استقلال خطی» یعنی «هر ترکیب خطی فقط یک نمایش دارد».
\end{note}

\subsection{زیرمجموعه‌ها و ابرمجموعه‌ها}
\begin{theorem}\label{thm:subset-superset}
\begin{enumerate}
  \item اگر مجموعه‌ای از بردارها وابسته باشد، هر \textbf{ابرمجموعه‌ی} آن هم وابسته است.
  \item هر زیرمجموعه‌ی ناتهی از یک مجموعه‌ی مستقل خطی، مستقل خطی است.
\end{enumerate}
\end{theorem}
\begin{proof}
\textbf{۱.} اگر $c_1\vv_1+\dots+c_k\vv_k=\zero$ رابطه‌ای با ضرایب نه‌همه‌صفر باشد، با اضافه کردن بردارهای جدید $\vv_{k+1},\dots,\vv_n$ با ضریب صفر، همچنان یک رابطه‌ی وابستگی (با ضرایب نه‌همه‌صفر) برای ابرمجموعه داریم.

\textbf{۲.} این همان بند ۱ است به شکل عکس نقیض: اگر یک زیرمجموعه وابسته بود، طبق بند ۱ کل مجموعه هم وابسته می‌شد.
\end{proof}

\subsection{جمع‌بندی روش بررسی}
برای بررسی استقلال $p$ بردار در $\R^n$:
\begin{enumerate}
  \item \textbf{بشمارید:} اگر $p>n$، مجموعه حتماً وابسته است. اگر $p\le n$ به قدم بعد بروید.
  \item \textbf{دنبال بردار صفر بگردید:} اگر هست، مجموعه وابسته است.
  \item \textbf{دو بردار؟} وابسته‌اند اگر و تنها اگر یکی مضرب دیگری باشد.
  \item \textbf{در حالت کلی:} بردارها را ستون‌های ماتریس $A$ قرار دهید و کاهش سطری کنید. مستقل‌اند اگر و تنها اگر \emph{همه‌ی} ستون‌ها محوری باشند.
\end{enumerate}
در فصل ۷ می‌بینیم که \textbf{رتبه‌ی} یک ماتریس دقیقاً بیشترین تعداد ستون‌های (یا سطرهای) مستقل خطی آن است.

\subsection{بردارهای جواب عمومی $A\vx=\zero$}
\begin{theorem}\label{thm:null-vectors-indep}
بردارهایی که در شکل برداری-پارامتری جواب عمومی $A\vx=\zero$ ظاهر می‌شوند (یک بردار برای هر متغیر آزاد) مستقل خطی‌اند.
\end{theorem}
\begin{proof}
فرض کنید متغیرهای آزاد $x_{f_1},\dots,x_{f_k}$ باشند و جواب عمومی
$\vx=x_{f_1}\vu_1+\dots+x_{f_k}\vu_k$
باشد. بردار $\vu_i$ از گذاشتن $x_{f_i}=1$ و بقیه‌ی متغیرهای آزاد برابر صفر به دست می‌آید؛ پس مؤلفه‌ی $f_i$-ام $\vu_i$ برابر ۱ و مؤلفه‌ی $f_j$-ام آن (برای $j\neq i$) صفر است. حالا اگر $c_1\vu_1+\dots+c_k\vu_k=\zero$، به مؤلفه‌ی $f_i$-ام دو طرف نگاه کنید: سمت چپ برابر $c_i$ است (فقط $\vu_i$ آن‌جا ناصفر است). پس $c_i=0$ برای همه‌ی $i$ها.
\end{proof}

\begin{example}
برای $A=\begin{bmatrix}1&-1&2\\-2&2&-4\end{bmatrix}$ جواب عمومی $A\vx=\zero$ را بیابید و درباره‌ی ستون‌های $A$ چه می‌توان گفت؟
\end{example}
\begin{solution}
با کاهش سطری $A\sim\begin{bmatrix}1&-1&2\\0&0&0\end{bmatrix}$؛ پس $x_1=x_2-2x_3$ و $x_2,x_3$ آزادند:
\[
\vx=\begin{bmatrix}x_1\\x_2\\x_3\end{bmatrix}=x_2\begin{bmatrix}1\\1\\0\end{bmatrix}+x_3\begin{bmatrix}-2\\0\\1\end{bmatrix}.
\]
\begin{itemize}
  \item بردارهای مربوط به $x_2$ و $x_3$، یعنی $(1,1,0)$ و $(-2,0,1)$، \textbf{مستقل} هستند (به مؤلفه‌های دوم و سوم نگاه کنید).
  \item ستون‌های $A$ مربوط به $x_2$ و $x_3$ (یعنی ستون‌های غیرمحوری) به ستون اول \textbf{وابسته‌اند}: $\vect{A}_2=-\vect{A}_1$ و $\vect{A}_3=2\vect{A}_1$.
  \item در نتیجه $\Span\{\vect{A}_1,\vect{A}_2,\vect{A}_3\}=\Span\{\vect{A}_1\}$.
\end{itemize}
\end{solution}

\section{استقلال خطی توابع}

در فضای توابع، «بردار صفر» تابع صفر است؛ پس تساوی با صفر یعنی تساوی \textbf{برای همه‌ی $t$ها}.

\begin{definition}[استقلال خطی توابع]
توابع $f(t)$ و $g(t)$ را \textbf{وابسته‌ی خطی} می‌نامیم اگر ثابت‌هایی مثل $c_1,c_2$ وجود داشته باشند که \emph{هر دو صفر نیستند} و
\[
c_1f(t)+c_2g(t)=0\qquad\text{\fa{برای همه‌ی $t$ها}}.
\]
در غیر این صورت $f$ و $g$ \textbf{مستقل خطی} هستند. (برای تعداد بیشتر توابع هم تعریف مشابه است.)
\end{definition}

\begin{example}
وابسته یا مستقل؟
\begin{enumerate}
  \item $f(t)=2\sin^2t$ و $g(t)=1-\cos^2t$.
  \item $\{\sin^2x,\ \cos^2x,\ \cos2x\}$.
  \item $\sin t$ و $\cos t$.
\end{enumerate}
\end{example}
\begin{solution}
\textbf{۱. وابسته.} از اتحاد $\sin^2t+\cos^2t=1$:
\[
f(t)-2g(t)=2\sin^2t-2+2\cos^2t=2-2=0\qquad\text{\fa{برای همه‌ی $t$}}.
\]
\textbf{۲. وابسته.} از اتحاد $\cos2x=\cos^2x-\sin^2x$:
\[
1\cdot\sin^2x-1\cdot\cos^2x+1\cdot\cos2x=-\cos2x+\cos2x=0 .
\]
\textbf{۳. مستقل.} فرض کنید $c_1\sin t+c_2\cos t=0$ برای \emph{همه‌ی} $t$ها. کافی است دو مقدار خوب برای $t$ بگذاریم:
\begin{itemize}
  \item $t=0$: \ $c_1\cdot0+c_2\cdot1=0$، پس $c_2=0$.
  \item $t=\frac\pi2$: \ $c_1\cdot1+c_2\cdot0=0$، پس $c_1=0$.
\end{itemize}
\end{solution}

\begin{note}[رونسکین]
روش دیگر برای توابع مشتق‌پذیر: اگر $c_1f+c_2g=0$ برای همه‌ی $t$، با مشتق گرفتن $c_1f'+c_2g'=0$. پس در هر نقطه‌ی $t_0$:
\[
\begin{bmatrix}f(t_0)&g(t_0)\\ f'(t_0)&g'(t_0)\end{bmatrix}\begin{bmatrix}c_1\\c_2\end{bmatrix}=\begin{bmatrix}0\\0\end{bmatrix}.
\]
اگر برای یک $t_0$ عدد $W(t_0)=f(t_0)g'(t_0)-g(t_0)f'(t_0)$ (\term{رونسکین}{Wronskian}) ناصفر باشد، این ماتریس $2\times2$ وارون‌پذیر است و تنها جواب $c_1=c_2=0$ است؛ پس $f,g$ مستقل‌اند. مثلاً برای $e^t$ و $e^{2t}$: $W(0)=1\cdot2-1\cdot1=1\neq0$.
\end{note}

\begin{warning}
برای اثبات \emph{وابستگی} توابع، باید رابطه را برای \textbf{همه‌ی} $t$ها نشان دهید (معمولاً با یک اتحاد). برای اثبات \emph{استقلال} کافی است چند مقدار $t$ پیدا کنید که ضرایب را مجبور به صفر شدن کنند. اما برقرار بودن رابطه در چند نقطه، وابستگی را ثابت \emph{نمی‌کند}.
\end{warning}

\section{استقلال خطی چندجمله‌ای‌ها}

\begin{example}
آیا $1-x$، $1+x$ و $x^2$ (در $P_2(\R)$) مستقل خطی‌اند؟
\end{example}
\begin{solution}
فرض کنید
\[
c_1(1-x)+c_2(1+x)+c_3x^2=0 .
\]
جمله‌های هم‌درجه را دسته‌بندی می‌کنیم:
\[
(c_1+c_2)+(-c_1+c_2)x+c_3x^2=0 .
\]
یک چندجمله‌ای صفر است اگر و تنها اگر همه‌ی ضرایبش صفر باشد؛ پس
\[
c_1+c_2=0,\qquad -c_1+c_2=0,\qquad c_3=0
\ \Longrightarrow\ c_1=c_2=c_3=0 .
\]
پس این سه چندجمله‌ای \textbf{مستقل خطی} هستند.
\end{solution}

\begin{theorem}\label{thm:monomials-indep}
برای هر $n$، چندجمله‌ای‌های $1,x,x^2,\dots,x^n$ (به عنوان توابعی روی $\R$) مستقل خطی‌اند.
\end{theorem}
\begin{proof}
فرض کنید $c_0+c_1x+\dots+c_nx^n=0$ برای \emph{همه‌ی} $x\in\R$. اگر همه‌ی $c_i$ها صفر نبودند، سمت چپ یک چندجمله‌ای ناصفر با درجه‌ی حداکثر $n$ بود؛ چنین چندجمله‌ای حداکثر $n$ ریشه دارد. اما این‌جا هر عدد حقیقی ریشه است (بی‌شمار ریشه). تناقض؛ پس همه‌ی $c_i$ها صفرند.
\end{proof}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item $\vv_1,\dots,\vv_n$ مستقل‌اند اگر $\sum\lambda_i\vv_i=\zero$ فقط جواب بدیهی داشته باشد؛ معادلاً هیچ برداری ترکیب خطی بقیه نیست.
  \item ستون‌های $A$ مستقل‌اند $\iff A\vx=\zero$ فقط جواب بدیهی دارد $\iff$ همه‌ی ستون‌ها محوری‌اند.
  \item مجموعه‌ی شامل $\zero$ وابسته است؛ بیش از $n$ بردار در $\R^n$ وابسته‌اند؛ دو بردار وابسته‌اند اگر هم‌راستا باشند.
  \item در مجموعه‌ی وابسته (با $\vv_1\neq\zero$) برداری هست که ترکیب خطی بردارهای قبلی است و می‌توان آن را بدون تغییر پوشش حذف کرد.
  \item نمایش هر بردار بر حسب بردارهای مستقل یکتاست.
  \item ابرمجموعه‌ی مجموعه‌ی وابسته وابسته است؛ زیرمجموعه‌ی مجموعه‌ی مستقل مستقل است.
  \item برای توابع، تساوی با صفر یعنی برای همه‌ی $t$ها؛ برای استقلال، مقدارگذاری یا رونسکین؛ برای وابستگی، یک اتحاد.
\end{itemize}
\end{summary}

\begin{exercise}
آیا $(1,2,3)$، $(4,5,6)$ و $(7,8,9)$ مستقل خطی‌اند؟ اگر نه، یک رابطه‌ی وابستگی بنویسید.
\end{exercise}

\begin{exercise}
به ازای چه مقدار $h$ بردارهای $(1,-1,4)$، $(3,-5,7)$ و $(-1,5,h)$ وابسته‌ی خطی‌اند؟
\end{exercise}

\begin{exercise}
ثابت کنید اگر $\{\vu,\vv\}$ مستقل خطی باشد، $\{\vu+\vv,\ \vu-\vv\}$ هم مستقل خطی است.
\end{exercise}

\begin{exercise}
آیا توابع $1$، $\cos2x$ و $\sin^2x$ مستقل خطی‌اند؟
\end{exercise}

\begin{exercise}
درست یا نادرست؟ «اگر $\vv_3$ ترکیب خطی $\vv_1$ و $\vv_2$ نباشد، آن‌گاه $\{\vv_1,\vv_2,\vv_3\}$ مستقل خطی است.»
\end{exercise}

\subsection*{پاسخ‌های کوتاه}
\begin{enumerate}
  \item وابسته‌اند: $(1,2,3)-2(4,5,6)+(7,8,9)=\zero$.
  \item با کاهش سطری ماتریسی که این‌ها ستون‌هایش هستند، درایه‌ی محوری سوم $h-6$ می‌شود؛ پس فقط برای $h=6$ وابسته‌اند.
  \item اگر $a(\vu+\vv)+b(\vu-\vv)=\zero$ آن‌گاه $(a+b)\vu+(a-b)\vv=\zero$؛ از استقلال $a+b=0$ و $a-b=0$، پس $a=b=0$.
  \item وابسته‌اند: $\cos2x=1-2\sin^2x$، یعنی $1\cdot1-1\cdot\cos2x-2\sin^2x=0$.
  \item نادرست: $\vv_1=(1,0)$، $\vv_2=(2,0)$، $\vv_3=(0,1)$.
\end{enumerate}
