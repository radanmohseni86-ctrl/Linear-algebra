\chapter{پایه و بُعد}
\label{ch:basis}

مجموعه‌ای که هم فضا را \textbf{تولید کند} و هم \textbf{مستقل} باشد، \emph{پایه} نام دارد. پایه مثل دستگاه مختصات فضاست: هر بردار به‌طور یکتا بر حسب آن نوشته می‌شود. همه‌ی پایه‌های یک فضا تعداد عضو یکسانی دارند و این عدد \emph{بُعد} فضاست.

در مسئله‌ی قیمت خانه، پرسش‌های این فصل این‌هاست: کوچک‌ترین مجموعه از ستون‌ها که همه‌ی اطلاعات را دارد چیست؟ (\textbf{پایه}) چند ویژگی واقعاً مستقل داریم؟ (\textbf{بُعد}) هر ستون دیگر را چطور بر حسب آن‌ها بنویسیم؟ (\textbf{مختصات})

\section{پایه}

\begin{definition}[پایه]
مجموعه‌ی $\mathcal{B}=\{\vb_1,\dots,\vb_n\}$ یک \term{پایه}{Basis} برای زیرفضای $H$ است اگر
(۱) $\mathcal{B}$ مستقل خطی باشد و (۲) $H=\Span\{\vb_1,\dots,\vb_n\}$.
\end{definition}

شرط دوم یعنی «به اندازه‌ی کافی بردار داریم» و شرط اول یعنی «بردار اضافه نداریم».

\begin{example}[پایه‌های استاندارد]
\begin{itemize}
  \item $\{\ve_1,\dots,\ve_n\}$ \term{پایه‌ی استاندارد}{Standard Basis} $\R^n$ است: $\vx=x_1\ve_1+\dots+x_n\ve_n$ و این ترکیب فقط وقتی صفر است که همه‌ی $x_i$ صفر باشند.
  \item $\{1,x,\dots,x^n\}$ پایه‌ی $P_n(\R)$ است (قضیه‌ی \ref{thm:monomials-indep}).
  \item ماتریس‌های $E_{ij}$ (۱ در جای $(i,j)$ و صفر در بقیه) پایه‌ی $M_{m\times n}$ هستند؛ مثلاً $\begin{bmatrix}a&b\\c&d\end{bmatrix}=aE_{11}+bE_{12}+cE_{21}+dE_{22}$.
\end{itemize}
\end{example}

\begin{warning}
پایه یکتا نیست: هر دو بردار غیرهم‌راستا در $\R^2$ یک پایه‌اند. اما وقتی پایه ثابت شد، \textbf{نمایش هر بردار یکتاست} (قضیه‌ی \ref{thm:unique-coeffs}).
\end{warning}

\begin{example}
آیا $\{1-x,\ 1+x,\ x^2\}$ پایه‌ای برای $P_2(\R)$ است؟
\end{example}
\begin{solution}
بله. استقلال را در فصل ۵ دیدیم. تولید:
$a+bx+cx^2=\frac{a-b}{2}(1-x)+\frac{a+b}{2}(1+x)+c\,x^2$.
\end{solution}

\begin{theorem}[قضیه‌ی مجموعه‌ی مولد]\label{thm:spanning-set}
اگر $S=\{\vv_1,\dots,\vv_p\}$ زیرفضای $H\neq\{\zero\}$ را تولید کند، زیرمجموعه‌ای از $S$ پایه‌ی $H$ است.
\end{theorem}
\begin{proof}
اگر $S$ وابسته باشد، طبق قضیه‌های \ref{thm:dep-char} و \ref{thm:remove-dependent} یک بردار را حذف می‌کنیم و پوشش عوض نمی‌شود. این کار را تا مستقل شدن تکرار می‌کنیم. چون $H\neq\{\zero\}$، همه‌ی بردارها حذف نمی‌شوند.
\end{proof}

\section{بُعد}

\begin{theorem}[مستقل‌ها حداکثر به تعداد مولدها]\label{thm:indep-le-span}
اگر $V$ توسط $\{\vx_1,\dots,\vx_m\}$ تولید شود، هر مجموعه‌ی مستقل در $V$ حداکثر $m$ عضو دارد.
\begin{center}
\hl{تعداد بردارهای مستقل \ $\le$ \ تعداد بردارهای مولد}
\end{center}
\end{theorem}
\begin{proof}
نشان می‌دهیم هر $\vy_1,\dots,\vy_n$ با $n>m$ وابسته‌اند. هر $\vy_i=\sum_{j=1}^m a_{ij}\vx_j$؛ پس
\[
c_1\vy_1+\dots+c_n\vy_n=\sum_{j=1}^m\Big(\sum_{i=1}^n a_{ij}c_i\Big)\vx_j .
\]
دستگاه همگن $\sum_i a_{ij}c_i=0$ ($j=1,\dots,m$) $m$ معادله و $n>m$ مجهول دارد و طبق نتیجه‌ی \ref{cor:more-unknowns} جواب نابدیهی دارد. با این $c_i$ها، $\sum c_i\vy_i=\zero$.
\end{proof}

\begin{theorem}[همه‌ی پایه‌ها هم‌اندازه‌اند]\label{thm:bases-same-size}
هر دو پایه‌ی یک فضای متناهی‌البُعد تعداد عضو یکسانی دارند.
\end{theorem}
\begin{proof}
اگر $\mathcal{B}_1$ با $m$ عضو و $\mathcal{B}_2$ با $n$ عضو دو پایه باشند، چون $\mathcal{B}_1$ مولد و $\mathcal{B}_2$ مستقل است، $n\le m$؛ و برعکس $m\le n$.
\end{proof}

\begin{definition}[بُعد]
$V$ \term{متناهی‌البُعد}{Finite-Dimensional} است اگر پایه‌ای متناهی داشته باشد؛ \term{بُعد}{Dimension} آن، $\dim V$، تعداد اعضای هر پایه است. در غیر این صورت $\dim V=\infty$. بُعد $\{\zero\}$ صفر است (پایه‌اش مجموعه‌ی تهی است).
\end{definition}

\begin{theorem}\label{thm:dim-bounds}
اگر $\dim V=m$، آن‌گاه (آ) هر مجموعه با بیش از $m$ بردار وابسته است؛ (ب) هیچ مجموعه با کمتر از $m$ بردار $V$ را تولید نمی‌کند.
\end{theorem}
\begin{proof}
(آ) مستقیماً از قضیه‌ی \ref{thm:indep-le-span}. (ب) اگر $k<m$ بردار $V$ را تولید کنند، طبق قضیه‌ی \ref{thm:spanning-set} پایه‌ای با حداکثر $k<m$ عضو داریم؛ تناقض.
\end{proof}

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lll}
\toprule
\textbf{فضا} & \textbf{یک پایه} & \textbf{بُعد}\\
\midrule
$\F^n$ & $\{\ve_1,\dots,\ve_n\}$ & $n$\\
$P_p$ & $\{1,x,\dots,x^p\}$ & $p+1$ \ \textcolor{warncolor}{(نه $p$!)}\\
$M_{m,n}$ & $\{E_{ij}\}$ & $mn$\\
$P$ (همه‌ی چندجمله‌ای‌ها) & $\{1,x,x^2,\dots\}$ & $\infty$\\
\bottomrule
\end{tabular}
\end{center}
$P$ نامتناهی‌البُعد است، چون برای هر $d$ مجموعه‌ی مستقل $\{1,x,\dots,x^d\}$ را دارد و این با قضیه‌ی \ref{thm:dim-bounds} برای هر بُعد متناهی تناقض دارد. فضای همه‌ی توابع و توابع پیوسته هم به همین دلیل نامتناهی‌البُعدند.

\section{زیرفضاهای فضای متناهی‌البُعد}

\begin{lemma}\label{lem:adjoin}
اگر $S$ مستقل باشد و $\vu\notin\Span(S)$، آن‌گاه $S\cup\{\vu\}$ مستقل است.
\end{lemma}
\begin{proof}
در $a_1\vv_1+\dots+a_n\vv_n+c\vu=\zero$، اگر $c\neq0$ آن‌گاه $\vu=-\frac1c\sum a_i\vv_i\in\Span(S)$؛ تناقض. پس $c=0$ و از استقلال $S$ همه‌ی $a_i$ صفرند.
\end{proof}

\begin{theorem}[گسترش به پایه]\label{thm:extend-basis}
اگر $W$ زیرفضای فضای متناهی‌البُعد $V$ باشد، هر زیرمجموعه‌ی مستقل $W$ متناهی است و به یک پایه‌ی $W$ گسترش می‌یابد.
\end{theorem}
\begin{proof}
طبق قضیه‌ی \ref{thm:dim-bounds} هر مجموعه‌ی مستقل حداکثر $\dim V$ عضو دارد. از بین مجموعه‌های مستقلِ $W$ که $S_0$ را دارند، بزرگ‌ترینش، $B$، را بگیرید. اگر $B$ مولد $W$ نباشد، $\vw\in W\setminus\Span(B)$ هست و طبق لم \ref{lem:adjoin}، $B\cup\{\vw\}$ مستقل و بزرگ‌تر است؛ تناقض.
\end{proof}

\begin{corollary}\label{cor:extend-in-V}
در فضای متناهی‌البُعد $V$، هر مجموعه‌ی مستقل بخشی از یک پایه‌ی $V$ است ($W=V$).
\end{corollary}

\begin{corollary}\label{cor:proper-subspace}
اگر $W$ زیرفضای \term{سره}{Proper} ($W\neq V$) فضای متناهی‌البُعد $V$ باشد، $\dim W<\dim V$.
\end{corollary}
\begin{proof}
پایه‌ی $B$ از $W$ در $V$ هم مستقل است؛ پس $\dim W\le\dim V$. اگر برابر بودند، برای $\vv\in V\setminus W$ طبق لم \ref{lem:adjoin}، $B\cup\{\vv\}$ مستقل با $\dim V+1$ عضو بود؛ تناقض.
\end{proof}

پس زیرفضاهای $\R^3$ دقیقاً این‌ها هستند: $\{\zero\}$ (بُعد ۰)، خط‌های گذرنده از مبدأ (بُعد ۱)، صفحه‌های گذرنده از مبدأ (بُعد ۲) و خود $\R^3$.

\begin{theorem}\label{thm:basis-props}
اگر $\dim V=n$:
\begin{enumerate}
  \item هر فهرست مستقل به یک پایه گسترش می‌یابد؛ هر فهرست مولد یک پایه در خود دارد.
  \item هر $n$ بردار \textbf{مستقل}، پایه است؛ و هر $n$ بردار \textbf{مولد}، پایه است.
\end{enumerate}
\end{theorem}
\begin{proof}
بند ۱ همان نتیجه‌ی \ref{cor:extend-in-V} و قضیه‌ی \ref{thm:spanning-set} است. برای بند ۲: گسترش یا کوچک‌کردن یک فهرست $n$تایی به پایه‌ای $n$تایی یعنی هیچ برداری اضافه یا حذف نشده است.
\end{proof}

\begin{note}
پس اگر بُعد را بدانیم، کافی است \emph{فقط یکی} از دو شرط پایه را بررسی کنیم. مثلاً چون $\dim P_2=3$، استقلال $\{1-x,1+x,x^2\}$ به‌تنهایی کافی بود.
\end{note}

\section{فرمول بُعد}

\begin{theorem}[فرمول بُعد]\label{thm:dim-formula}
برای زیرفضاهای متناهی‌البُعد $W_1,W_2$:
\begin{keyformula}
\[
\dim W_1+\dim W_2=\dim(W_1\cap W_2)+\dim(W_1+W_2).
\]
\end{keyformula}
\end{theorem}
\begin{proof}
پایه‌ی $\{\vv_1,\dots,\vv_m\}$ از $W_1\cap W_2$ را به پایه‌ی $\{\vv_i,\vu_1,\dots,\vu_j\}$ از $W_1$ و پایه‌ی $\{\vv_i,\vw_1,\dots,\vw_k\}$ از $W_2$ گسترش دهید. نشان می‌دهیم $\mathcal{B}=\{\vv_i,\vu_i,\vw_i\}$ پایه‌ی $W_1+W_2$ است؛ آن‌گاه $\dim(W_1+W_2)=m+j+k=(m+j)+(m+k)-m$.

\textbf{تولید:} هر $\vx_1+\vx_2$ با $\vx_1\in W_1$ و $\vx_2\in W_2$ ترکیب خطی اعضای $\mathcal{B}$ است.

\textbf{استقلال:} اگر $\sum a_i\vv_i+\sum b_i\vu_i+\sum c_i\vw_i=\zero$، بردار $\sum c_i\vw_i=-\sum a_i\vv_i-\sum b_i\vu_i$ هم در $W_2$ است و هم در $W_1$؛ پس $\sum c_i\vw_i=\sum d_i\vv_i$. چون $\{\vv_i,\vw_i\}$ مستقل است، همه‌ی $c_i$ صفرند؛ و سپس چون $\{\vv_i,\vu_i\}$ مستقل است، $a_i$ و $b_i$ هم صفرند.
\end{proof}

\begin{intuition}
شبیه $|A\cup B|=|A|+|B|-|A\cap B|$ است: بخش مشترک دو بار شمرده می‌شود. نتیجه: $\dim(U\oplus W)=\dim U+\dim W$.
\end{intuition}

\begin{example}
نشان دهید دو صفحه‌ی گذرنده از مبدأ در $\R^3$ دست‌کم در یک خط مشترک‌اند.
\end{example}
\begin{solution}
$\dim(W_1\cap W_2)=2+2-\dim(W_1+W_2)\ge4-3=1$.
\end{solution}

\begin{warning}[سه زیرفضا]
فرمول شمول و عدم شمول برای سه زیرفضا برقرار نیست: سه خط $\Span\{(1,0)\}$، $\Span\{(0,1)\}$ و $\Span\{(1,1)\}$ در $\R^2$ را در نظر بگیرید. همه‌ی اشتراک‌ها $\{\zero\}$ است، پس سمت راست $1+1+1=3$ می‌شود، اما $\dim(W_1+W_2+W_3)=2$. (همیشه فقط «$\le$» برقرار است.)
\end{warning}

\section{مختصات}

\begin{definition}[بردار مختصات]
اگر $\mathcal{B}=\{\vb_1,\dots,\vb_p\}$ یک \term{پایه‌ی مرتب}{Ordered Basis} برای $H$ باشد و $\vx=c_1\vb_1+\dots+c_p\vb_p$، بردار
$\coord{\vx}{\mathcal{B}}=(c_1,\dots,c_p)$
\term{بردار مختصات}{Coordinate Vector} $\vx$ نسبت به $\mathcal{B}$ است. ترتیب بردارهای پایه، ترتیب مؤلفه‌ها را تعیین می‌کند.
\end{definition}

\begin{example}
در $\R^2$ با $\vb_1=(2,1)$ و $\vb_2=(-1,1)$، $\coord{\vx}{\mathcal{B}}$ را برای $\vx=(4,5)$ بیابید.
\end{example}
\begin{solution}
از $2c_1-c_2=4$ و $c_1+c_2=5$: $c_1=3,\ c_2=2$. پس $\coord{\vx}{\mathcal{B}}=(3,2)$، در حالی که مختصات استاندارد $(4,5)$ است:

\begin{theorem}[نگاشت مختصات]\label{thm:coord-map}
$\coord{\vu+\vv}{\mathcal{B}}=\coord{\vu}{\mathcal{B}}+\coord{\vv}{\mathcal{B}}$ و $\coord{c\vu}{\mathcal{B}}=c\coord{\vu}{\mathcal{B}}$. به‌علاوه $\vu_1,\dots,\vu_k$ مستقل‌اند اگر و تنها اگر بردارهای مختصاتشان مستقل باشند.
\end{theorem}
\begin{proof}
اگر $\vu=\sum a_i\vb_i$ و $\vv=\sum b_i\vb_i$، آن‌گاه $\vu+\vv=\sum(a_i+b_i)\vb_i$ و $c\vu=\sum ca_i\vb_i$ و نمایش یکتاست. پس $\sum c_i\vu_i=\zero$ دقیقاً وقتی است که $\sum c_i\coord{\vu_i}{\mathcal{B}}=\zero$.
\end{proof}

\begin{note}
پس هر فضای $n$-بعدی از نظر جبر خطی «همان» $\F^n$ است؛ مثلاً برای بررسی استقلال چندجمله‌ای‌ها کافی است بردار ضرایبشان را کاهش سطری کنیم.
\end{note}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item پایه = مستقل + مولد. هر مجموعه‌ی مولد متناهی یک پایه در خود دارد و هر مجموعه‌ی مستقل به پایه گسترش می‌یابد.
  \item تعداد مستقل‌ها $\le$ تعداد مولدها؛ پس همه‌ی پایه‌ها هم‌اندازه‌اند و $\dim V$ خوش‌تعریف است.
  \item در فضای $n$-بعدی: $n$ بردار مستقل یا $n$ بردار مولد، پایه‌اند؛ زیرفضای سره بُعد کمتری دارد.
  \item $\dim W_1+\dim W_2=\dim(W_1\cap W_2)+\dim(W_1+W_2)$.
  \item مختصات نسبت به پایه‌ی مرتب یکتاست و با جمع و ضرب اسکالر سازگار است.
\end{itemize}
\end{summary}

\subsection*{تمرین‌ها}
\begin{enumerate}
  \item یک پایه و بُعد $W=\{(x,y,z): x+y+z=0\}$ را بیابید.
  \item بُعد فضای ماتریس‌های متقارن $2\times2$ چیست؟
  \item مختصات $1+2x+3x^2$ را نسبت به پایه‌ی $\{1-x,\ 1+x,\ x^2\}$ بیابید.
  \item اگر $\dim V=5$ و $W_1,W_2$ زیرفضاهای سه‌بعدی باشند، نشان دهید $W_1\cap W_2\neq\{\zero\}$.
\end{enumerate}

\subsection*{پاسخ‌های کوتاه}
\begin{enumerate}
  \item $x=-y-z$؛ پایه $\{(-1,1,0),(-1,0,1)\}$ و $\dim W=2$.
  \item ۳؛ پایه $\{E_{11},\,E_{12}+E_{21},\,E_{22}\}$.
  \item با $a=1,b=2,c=3$ در فرمول بالا: $(-\frac12,\ \frac32,\ 3)$.
  \item $\dim(W_1\cap W_2)\ge3+3-5=1$.
\end{enumerate}
