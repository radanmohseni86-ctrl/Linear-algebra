\chapter{پیش‌درآمد: نمادهای پایه}
\label{ch:prelim}

این فصل کوتاه چند نماد و مفهوم را مرور می‌کند که در ادامه زیاد به کار می‌روند. اگر با آن‌ها آشنا هستید، مستقیم به فصل ۱ بروید.

\section{نماد سیگما}
$\sum_{i=1}^{n} a_i:=a_1+\dots+a_n$. سه قاعده‌ی پرکاربرد:
\[
\sum_{i} c\,a_i = c\sum_{i} a_i,\qquad
\sum_{i} (a_i+b_i) = \sum_{i} a_i+\sum_{i} b_i,\qquad
\sum_{i}\sum_{j} a_{ij} = \sum_{j}\sum_{i} a_{ij}.
\]
قاعده‌ی سوم یعنی در جمع دوگانه ترتیب جمع‌زدن مهم نیست.

\section{اعداد مختلط}\label{sec:complex}
عدد مختلط $z=a+bi$ است که $a,b\in\R$ و $i^2=-1$. \term{مزدوج}{Conjugate} آن $\conj{z}=a-bi$ و \term{قدرمطلق}{Modulus} آن $\abs{z}=\sqrt{a^2+b^2}$ است. ویژگی‌هایی که در فصل ۸ لازم داریم:
\[
z\conj{z}=\abs{z}^2\ge 0,\qquad \conj{z+w}=\conj{z}+\conj{w},\qquad \conj{zw}=\conj{z}\,\conj{w},\qquad \operatorname{Re}(z)\le\abs{z}.
\]

\section{میدان}\label{sec:field}
\begin{definition}[میدان]
\term{میدان}{Field} مجموعه‌ای مثل $\F$ با دو عمل جمع و ضرب است که در آن این چهار عمل بدون مشکل انجام می‌شوند: جمع و ضرب بسته، جابه‌جایی، شرکت‌پذیر و توزیع‌پذیرند؛ عضوهای $0$ و $1$ وجود دارند؛ هر عضو قرینه دارد و هر عضو ناصفر وارون ضربی.
\end{definition}

مثلاً $\Q$، $\R$ و $\C$ میدان‌اند، اما $\Z$ نه (۲ وارون ضربی در $\Z$ ندارد). \textbf{در این درس اگر چیزی نگوییم، میدان $\R$ است.}

\textbf{دلتای کرونکر:} $\delta_{ij}=1$ اگر $i=j$ و در غیر این صورت $0$. مثلاً درایه‌ی $(i,j)$ ماتریس همانی $I$ برابر $\delta_{ij}$ است.
