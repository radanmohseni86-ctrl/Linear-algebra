\chapter{فضای برداری}
\label{ch:vector-spaces}

بردارهای $\R^n$ را جمع می‌زنیم و در عدد ضرب می‌کنیم. ماتریس‌ها، چندجمله‌ای‌ها و توابع هم همین رفتار را دارند و همان قواعد آشنا در آن‌ها برقرار است. ایده‌ی \textbf{فضای برداری} این است که این قواعد مشترک را جدا کنیم و یک بار برای همه‌ی آن‌ها حرف بزنیم.

\section{عمل دوتایی و بستاری}

\begin{definition}[عمل دوتایی و بسته بودن]
هر تابع $*:A\times A\to A$ یک \term{عمل دوتایی}{Binary Operation} روی $A$ است. مجموعه‌ی $A$ تحت یک عمل \term{بسته}{Closed} است اگر حاصل عمل روی اعضای $A$ همیشه در $A$ بماند: $a,b\in A\Rightarrow a*b\in A$ (\term{قانون بستاری}{Closure Law}).
\end{definition}

مثلاً $\N$ تحت جمع بسته است اما تحت تفریق نه ($2-3\notin\N$)؛ $\Z$ تحت تقسیم بسته نیست و اعداد فرد تحت جمع بسته نیستند ($1+1=2$).

\section{تعریف فضای برداری}

\begin{definition}[فضای برداری]
یک \term{فضای برداری}{Vector Space} روی میدان $\F$ (پیش‌فرض: $\R$)، مجموعه‌ی ناتهی $V$ است با دو عمل
\textbf{جمع برداری} $\vx+\vy\in V$ و \textbf{ضرب اسکالر} $\alpha\vx\in V$ ($\alpha\in\F$)،
به‌طوری که برای همه‌ی $\vx,\vy,\vz\in V$ و $\alpha,\beta\in\F$:
\begin{multicols}{2}
\begin{enumerate}
  \item $\vx+\vy=\vy+\vx$
  \item $(\vx+\vy)+\vz=\vx+(\vy+\vz)$
  \item $\zero\in V$ هست که $\vx+\zero=\vx$
  \item $-\vx\in V$ هست که $\vx+(-\vx)=\zero$
  \item $\alpha(\beta\vx)=(\alpha\beta)\vx$
  \item $(\alpha+\beta)\vx=\alpha\vx+\beta\vx$
  \item $\alpha(\vx+\vy)=\alpha\vx+\alpha\vy$
  \item $1\vx=\vx$
\end{enumerate}
\end{multicols}
\end{definition}

\begin{intuition}
لازم نیست بردارها پیکان باشند؛ هر چیزی که بتوان جمعش زد و در عدد ضربش کرد و این هشت قانون را رعایت کند، بردار است. هر قضیه‌ای که فقط با این اصول ثابت شود، هم‌زمان برای $\R^n$، ماتریس‌ها، چندجمله‌ای‌ها و توابع درست است.
\end{intuition}

\begin{warning}
در فضای برداری کلی \textbf{ضرب دو بردار} تعریف نشده و طول و زاویه هم معنی ندارد. این‌ها را در فصل ۸ با ضرب داخلی اضافه می‌کنیم.
\end{warning}

\section{نتایج اولیه‌ی اصول}

\begin{theorem}[ویژگی‌های پایه]\label{thm:vs-basic}
در هر فضای برداری، برای هر $\vx,\vy,\vz\in V$ و $\alpha\in\F$:
\begin{enumerate}
  \item \textbf{قانون حذف:} اگر $\vx+\vy=\vx+\vz$ آن‌گاه $\vy=\vz$.
  \item بردار صفر و قرینه‌ی هر بردار یکتاست.
  \item $0\vx=\zero$ و $\alpha\zero=\zero$.
  \item $(-1)\vx=-\vx$.
  \item اگر $\alpha\vx=\zero$ آن‌گاه $\alpha=0$ یا $\vx=\zero$.
\end{enumerate}
\end{theorem}
\begin{proof}
\textbf{۱.} $-\vx$ را از چپ به دو طرف اضافه می‌کنیم:
$\vy=\big((-\vx)+\vx\big)+\vy=(-\vx)+(\vx+\vy)=(-\vx)+(\vx+\vz)=\vz$.

\textbf{۲.} اگر $\zero$ و $\zero'$ هر دو خنثی باشند، $\zero'=\zero'+\zero=\zero+\zero'=\zero$. اگر $\vy$ و $\vz$ هر دو قرینه‌ی $\vx$ باشند، $\vx+\vy=\zero=\vx+\vz$ و طبق قانون حذف $\vy=\vz$.

\textbf{۳.} $0\vx+\zero=0\vx=(0+0)\vx=0\vx+0\vx$ و طبق قانون حذف $0\vx=\zero$. به همین شکل $\alpha\zero=\alpha(\zero+\zero)=\alpha\zero+\alpha\zero$ و $\alpha\zero=\zero$.

\textbf{۴.} $\vx+(-1)\vx=(1+(-1))\vx=0\vx=\zero$؛ پس $(-1)\vx$ قرینه‌ی $\vx$ است و چون قرینه یکتاست، برابر $-\vx$ است.

\textbf{۵.} اگر $\alpha\neq0$، وارون $\alpha^{-1}$ در میدان وجود دارد و
$\vx=1\vx=\alpha^{-1}(\alpha\vx)=\alpha^{-1}\zero=\zero$.
\end{proof}

\section{نمونه‌هایی از فضای برداری}

\begin{itemize}
  \item \textbf{$\R^n$ و $\F^n$:} با جمع و ضرب مؤلفه‌به‌مؤلفه. اصول مستقیماً از ویژگی‌های اعداد می‌آیند. صفر: $(0,\dots,0)$.
  \item \textbf{ماتریس‌ها $M_{m,n}(\R)$:} هر ماتریس $mn$ عدد است که درایه‌به‌درایه جمع و ضرب می‌شوند؛ پس مثل $\R^{mn}$ است.
  \item \textbf{توابع $\F^X$:} همه‌ی توابع $f:X\to\F$ با عمل‌های \textbf{نقطه‌به‌نقطه}
  $(f+g)(x)=f(x)+g(x)$ و $(af)(x)=af(x)$. هر اصل را کافی است در یک نقطه‌ی دلخواه $x$ بررسی کنیم، و آن‌جا همه‌چیز به ویژگی‌های $\F$ برمی‌گردد. صفر، تابع صفر است و $(-f)(x)=-f(x)$. دنباله‌ها هم توابعی از $\N$ به $\R$ هستند.
  \item \textbf{چندجمله‌ای‌ها:} $P(\R)$ همه‌ی چندجمله‌ای‌ها و $P_n(\R)$ چندجمله‌ای‌های با درجه‌ی \textbf{حداکثر} $n$. جمع و ضرب روی ضرایب انجام می‌شود؛ پس $a_0+\dots+a_nx^n$ مثل بردار $(a_0,\dots,a_n)\in\R^{n+1}$ رفتار می‌کند.
  \item توابع پیوسته‌ی $C[0,1]$ و توابع مشتق‌پذیر هم فضای برداری‌اند.
  \item \textbf{فضای صفر} $\{\zero\}$ کوچک‌ترین فضای برداری است.
\end{itemize}

\begin{warning}
چندجمله‌ای‌های با درجه‌ی \textbf{دقیقاً} $n$ فضای برداری نیستند: چندجمله‌ای صفر در آن‌ها نیست و تحت جمع هم بسته نیستند: $(x^2+x)+(-x^2)=x$.
\end{warning}

\begin{example}[کدام‌ها فضای برداری‌اند؟]
با عمل‌های معمولی: (الف) $\C$ روی $\R$؛ (ب) $\R$ روی $\C$؛ (ج) $\Z$ روی $\R$.
\end{example}
\begin{solution}
\textbf{(الف) بله؛} $a+bi$ مثل $(a,b)\in\R^2$ رفتار می‌کند.
\textbf{(ب) خیر؛} ضرب اسکالر بسته نیست: $i\cdot1=i\notin\R$.
\textbf{(ج) خیر؛} $\frac12\cdot1\notin\Z$.
\end{solution}

\begin{example}[عمل‌های عجیب]
(الف) $V=\R$ با $u\oplus v=u-v$ و ضرب معمولی. (ب) $V=\R^{+}$ (اعداد مثبت) با $u\oplus v=uv$ و $\alpha\odot u=u^\alpha$. کدام فضای برداری است؟
\end{example}
\begin{solution}
\textbf{(الف) خیر.} جابه‌جایی برقرار نیست: $1\oplus0=1$ اما $0\oplus1=-1$. توزیع‌پذیری روی جمع اسکالرها هم نقض می‌شود: $(1+1)\odot1=2$ اما $(1\odot1)\oplus(1\odot1)=0$.

\textbf{(ب) بله.} ضرب و توان اعداد مثبت، مثبت است؛ ضرب جابه‌جایی و شرکت‌پذیر است.
\textbf{«بردار صفر» عدد ۱ است} ($u\oplus1=u$) و قرینه‌ی $u$ عدد $\frac1u$ است. بقیه‌ی اصول همان قوانین توان‌اند:
$(u^\beta)^\alpha=u^{\alpha\beta}$،\ $u^{\alpha+\beta}=u^\alpha u^\beta$،\ $(uv)^\alpha=u^\alpha v^\alpha$ و $u^1=u$.
پس «صفرِ» یک فضا لزوماً عدد صفر نیست.
\end{solution}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item فضای برداری: مجموعه‌ای بسته تحت جمع و ضرب اسکالر که هشت اصل در آن برقرار است.
  \item نتایج: صفر و قرینه یکتا هستند، $0\vx=\zero$، $\alpha\zero=\zero$، $(-1)\vx=-\vx$ و اگر $\alpha\vx=\zero$ آن‌گاه $\alpha=0$ یا $\vx=\zero$.
  \item مثال‌ها: $\R^n$، $M_{m,n}$، $\F^X$، $P(\R)$، $P_n(\R)$ و توابع پیوسته.
  \item برای رد کردن، یک مثال نقض برای یک اصل (یا بستاری) کافی است.
\end{itemize}
\end{summary}

\subsection*{تمرین‌ها}
\begin{enumerate}
  \item آیا $\{(x,y)\in\R^2: x\ge0\}$ با عمل‌های معمولی فضای برداری است؟
  \item روی $\R^2$ با جمع معمولی و $c\odot(x,y)=(cx,0)$، کدام اصل برقرار نیست؟
  \item ثابت کنید اگر $\alpha\vx=\beta\vx$ و $\vx\neq\zero$، آن‌گاه $\alpha=\beta$.
\end{enumerate}

\subsection*{پاسخ‌های کوتاه}
\begin{enumerate}
  \item خیر؛ $(-1)(1,0)=(-1,0)$ در مجموعه نیست.
  \item فقط اصل همانی: $1\odot(x,y)=(x,0)\neq(x,y)$ وقتی $y\neq0$.
  \item $(\alpha-\beta)\vx=\zero$ و $\vx\neq\zero$؛ طبق بند ۵ قضیه‌ی \ref{thm:vs-basic}، $\alpha=\beta$.
\end{enumerate}
