\chapter{فرم‌های خطی، ضرب داخلی و نُرم}
\label{ch:inner-product}

در فضای برداری کلی از \emph{طول}، \emph{فاصله} و \emph{زاویه} خبری نیست. در این فصل این مفاهیم را قدم‌به‌قدم اضافه می‌کنیم:
\begin{center}
\hl{فرم خطی \ $\to$ \ فرم دوخطی \ $\to$ \ ضرب داخلی \ $\to$ \ نُرم (طول) \ $\to$ \ متر (فاصله)}
\end{center}

\section{فرم خطی}

\begin{definition}[تابع خطی]
تابع $f:\R^n\to\R$ \term{خطی}{Linear} است اگر \textbf{جمع‌پذیر} \enterm{Additive}، یعنی $f(\vx+\vy)=f(\vx)+f(\vy)$، و \textbf{همگن} \enterm{Homogeneous}، یعنی $f(\alpha\vx)=\alpha f(\vx)$، باشد. معادلاً ویژگی \term{برهم‌نهی}{Superposition} را دارد:
\[
f(\alpha\vx+\beta\vy)=\alpha f(\vx)+\beta f(\vy).
\]
تابع خطی از یک فضای برداری به میدانش را \term{فرم خطی}{Linear Form} می‌نامیم.
\end{definition}

\begin{theorem}\label{thm:inner-is-linear}
برای بردار ثابت $\va$، تابع $f(\vx)=\va\T\vx$ خطی است.
\end{theorem}
\begin{proof}
$\va\T(\alpha\vx+\beta\vy)=\sum a_i(\alpha x_i+\beta y_i)=\alpha\sum a_ix_i+\beta\sum a_iy_i=\alpha\va\T\vx+\beta\va\T\vy$.
\end{proof}

\begin{theorem}\label{thm:linear-is-inner}
برعکس، هر تابع خطی $f:\R^n\to\R$ به شکل $f(\vx)=\va\T\vx$ است، با $\va=\big(f(\ve_1),\dots,f(\ve_n)\big)$ که \textbf{یکتاست}.
\end{theorem}
\begin{proof}
$f(\vx)=f(x_1\ve_1+\dots+x_n\ve_n)=x_1f(\ve_1)+\dots+x_nf(\ve_n)=\va\T\vx$. اگر $\va\T\vx=\vb\T\vx$ برای همه‌ی $\vx$، با $\vx=\ve_i$ داریم $a_i=b_i$.
\end{proof}

\begin{example}
آیا «میانگین» خطی است؟ «بیشینه» چطور؟
\end{example}
\begin{solution}
\textbf{میانگین خطی است:} $\operatorname{avg}(\vx)=\frac1nx_1+\dots+\frac1nx_n=(\frac1n,\dots,\frac1n)\T\vx$.
\textbf{بیشینه نه:} برای $\vx=(5,4,0)$ و $\vy=(0,7,8)$، $\max(\vx+\vy)=11\neq5+8$.
\end{solution}

\begin{warning}
تابع \emph{آفین} $f(\vx)=\va\T\vx+b$ با $b\neq0$ خطی نیست، چون هر تابع خطی $f(\zero)=0$ دارد.
\end{warning}

\section{فرم دوخطی}

\begin{definition}[فرم دوخطی]
$f:V\times W\to\F$ یک \term{فرم دوخطی}{Bilinear Form} است اگر در هر آرگومان (با ثابت نگه داشتن دیگری) خطی باشد:
\[
f(\vv_1+c\vv_2,\vw)=f(\vv_1,\vw)+c\,f(\vv_2,\vw),\qquad
f(\vv,\vw_1+c\vw_2)=f(\vv,\vw_1)+c\,f(\vv,\vw_2).
\]
اگر $f(\vv,\vw)=f(\vw,\vv)$، فرم \term{متقارن}{Symmetric} است.
\end{definition}

\begin{example}[فضای دوگان]
\term{دوگان}{Dual} $V^*$ فضای همه‌ی فرم‌های خطی روی $V$ است. تابع $g(f,\vv)=f(\vv)$ روی $V^*\times V$ دوخطی است: خطی بودن در $f$ از تعریف جمع توابع ($(f_1+cf_2)(\vv)=f_1(\vv)+cf_2(\vv)$) و خطی بودن در $\vv$ از خطی بودن $f$ می‌آید.
\end{example}

\begin{theorem}\label{thm:bilinear-matrix}
هر فرم دوخطی روی $\R^n\times\R^n$ به شکل $f(\vv,\vw)=\vv\T A\vw$ با $A_{ij}=f(\ve_i,\ve_j)$ است؛ و $f$ متقارن است اگر و تنها اگر $A\T=A$.
\end{theorem}
\begin{proof}
$f\big(\sum_iv_i\ve_i,\sum_jw_j\ve_j\big)=\sum_i\sum_jv_iw_jf(\ve_i,\ve_j)=\vv\T A\vw$. اگر $f$ متقارن باشد $a_{ij}=a_{ji}$؛ و اگر $A\T=A$، چون $\vv\T A\vw$ عدد است، $\vv\T A\vw=(\vv\T A\vw)\T=\vw\T A\vv$.
\end{proof}

\begin{definition}[ماتریس وابسته]
نسبت به پایه‌ی $\mathcal{B}=\{\vb_1,\dots,\vb_n\}$، \term{ماتریس وابسته}{Associated Matrix} $f$ ماتریس $[f]_{\mathcal{B}}$ با درایه‌ی $(i,j)$ برابر $f(\vb_i,\vb_j)$ است و
$f(\vv,\vw)=\coord{\vv}{\mathcal{B}}\T[f]_{\mathcal{B}}\coord{\vw}{\mathcal{B}}$.
اگر $P=\begin{bmatrix}\vb_1&\cdots&\vb_n\end{bmatrix}$ و $A$ ماتریس استاندارد باشد، $[f]_{\mathcal{B}}=P\T AP$.
\end{definition}

\begin{example}
برای $f\big((a,b),(c,d)\big)=2ac+4ad-bc$، ماتریس $f$ را نسبت به پایه‌ی استاندارد و $\mathcal{B}=\{(2,1),(-1,4)\}$ بیابید.
\end{example}
\begin{solution}
$[f]_{\text{\fa{استاندارد}}}=\begin{bmatrix}2&4\\-1&0\end{bmatrix}$. با $\vb_1=(2,1)$ و $\vb_2=(-1,4)$:
$f(\vb_1,\vb_1)=8+8-2=14$،\
$f(\vb_1,\vb_2)=-4+32+1=29$،\
$f(\vb_2,\vb_1)=-4-4-8=-16$،\
$f(\vb_2,\vb_2)=2-16+4=-10$. پس
$[f]_{\mathcal{B}}=\begin{bmatrix}14&29\\-16&-10\end{bmatrix}$.\footnote{در اسلایدها درایه‌ی $(2,2)$ عدد $20$ نوشته شده بود؛ محاسبه (و $P\T AP$) عدد $-10$ می‌دهد.}
\end{solution}

\subsection*{روی اعداد مختلط}
با $f(\vv,\vw)=\vv\T\vw$ روی $\C$ داریم $f(i,i)=-1<0$؛ پس برای ساختن «طول» در آرگومان دوم \emph{مزدوج} می‌گیریم. فرم \term{یک‌ونیم‌خطی}{Sesquilinear} در آرگومان اول خطی و در دومی \textbf{مزدوج-خطی} است:
$f(\vv,\lambda\vw)=\textcolor{warncolor}{\conj{\lambda}}f(\vv,\vw)$.

\section{ضرب داخلی}

\begin{definition}[ضرب داخلی]
برای $\F=\R$ یا $\C$، \term{ضرب داخلی}{Inner Product} روی $V$ تابعی $\inner{\cdot}{\cdot}:V\times V\to\F$ است که:
\begin{enumerate}[label=(\alph*)]
  \item $\inner{\vv}{\vw}=\conj{\inner{\vw}{\vv}}$ \hfill (تقارن مزدوج؛ روی $\R$ همان تقارن)
  \item $\inner{\vv+c\vx}{\vw}=\inner{\vv}{\vw}+c\inner{\vx}{\vw}$ \hfill (خطی در آرگومان اول)
  \item $\inner{\vv}{\vv}\ge0$ و تساوی فقط برای $\vv=\zero$ \hfill (مثبت معین)
\end{enumerate}
روی $\R$ یعنی یک فرم دوخطیِ متقارنِ مثبت معین. فضای متناهی‌البُعد با ضرب داخلی را \term{فضای ضرب داخلی}{Inner Product Space} می‌نامیم (حقیقی: \term{اقلیدسی}{Euclidean}؛ مختلط: \term{یکانی}{Unitary}).
\end{definition}

\begin{note}
از (آ) نتیجه می‌شود $\inner{\vv}{\vv}$ حقیقی است و $\inner{\vv}{c\vw}=\conj{c}\inner{\vv}{\vw}$. روی $\R$ تقارن، خطی بودن در آرگومان دوم را هم می‌دهد. همچنین $\inner{\zero}{\vv}=\inner{0\vu}{\vv}=0$.
\end{note}

\begin{example}[نمونه‌ها]
\begin{itemize}
  \item $\R^n$: $\inner{\vx}{\vy}=\vx\T\vy$ (قضیه‌ی \ref{thm:dot-props}). \ $\C^n$: $\inner{\vv}{\vw}=\sum v_i\conj{w_i}$ و $\inner{\vv}{\vv}=\sum\abs{v_i}^2$.
  \item ماتریس‌ها: $\inner{X}{Y}=\tr(X\T Y)=\sum_{i,j}X_{ij}Y_{ij}$؛ یعنی ضرب داخلی معمولی اگر ماتریس را یک بردار بلند ببینیم.
  \item توابع پیوسته روی $[a,b]$: $\inner{f}{g}=\int_a^bfg\,dx$. اگر $\int f^2=0$ ولی $f(x_0)\neq0$، به خاطر پیوستگی $f^2$ در یک بازه‌ی کوچک اطراف $x_0$ مثبت است و انتگرال مثبت می‌شود؛ پس $f\equiv0$.
\end{itemize}
\end{example}

\begin{warning}[دو قرارداد]
اسلایدها روی $\C^n$ از $\vv^*\vw=\sum\conj{v_i}w_i$ استفاده می‌کنند که در آرگومان \emph{دوم} خطی است. هر دو قرارداد رایج و معتبرند؛ این جزوه «خطی در آرگومان اول» را به کار می‌برد.
\end{warning}

\begin{example}
(الف) $\inner{U}{V}$ را برای $U=\begin{bmatrix}1&2\\1&0\end{bmatrix}$ و $V=\begin{bmatrix}1&-1\\1&1\end{bmatrix}$ حساب کنید.
(ب) آیا $\inner{p}{q}=p(0)q(0)+\dots+p(n)q(n)$ روی $P_n$ ضرب داخلی است؟
\end{example}
\begin{solution}
(الف) $1\cdot1+2\cdot(-1)+1\cdot1+0\cdot1=0$؛ پس $U$ و $V$ بر هم عمودند.

(ب) بله. تقارن و خطی بودن روشن است. اگر $\inner{p}{p}=\sum p(k)^2=0$، چندجمله‌ای $p$ با درجه‌ی حداکثر $n$ دارای $n+1$ ریشه است؛ پس $p=0$. (روی $P_{n+1}$ ضرب داخلی نیست: $x(x-1)\cdots(x-n)$ ناصفر است ولی $\inner{p}{p}=0$.)
\end{solution}

\section{نُرم}

نُرم اندازه‌ی یک بردار را می‌سنجد و فاصله، نزدیکی دو بردار را؛ مثلاً در یادگیری ماشین برای سنجیدن شباهت دو تصویر.

\begin{definition}[نُرم]
\term{نُرم}{Norm} تابعی $\norm{\cdot}:V\to\R$ است که:
\begin{enumerate}
  \item $\norm{\alpha\vx}=\abs{\alpha}\norm{\vx}$ (همگنی مطلق)؛
  \item $\norm{\vx+\vy}\le\norm{\vx}+\norm{\vy}$ (نامساوی مثلث)؛
  \item $\norm{\vx}=0\Rightarrow\vx=\zero$.
\end{enumerate}
نامنفی بودن از این‌ها نتیجه می‌شود: $0=\norm{\vx-\vx}\le2\norm{\vx}$.
\end{definition}

\begin{theorem}[نامساوی کوشی-شوارتز]\label{thm:cauchy-schwarz}
با $\norm{\vx}=\sqrt{\inner{\vx}{\vx}}$:
\begin{keyformula}
\[
\abs{\inner{\vx}{\vy}}\le\norm{\vx}\,\norm{\vy},
\]
\end{keyformula}
و تساوی فقط وقتی است که یکی مضرب دیگری باشد.
\end{theorem}
\begin{proof}
اگر $\vy=\zero$ بدیهی است. وگرنه با $\lambda=\frac{\inner{\vx}{\vy}}{\norm{\vy}^2}$ و $\vz=\vx-\lambda\vy$ («سایه‌ی $\vx$ روی $\vy$» را کم می‌کنیم):
\[
0\le\inner{\vz}{\vz}=\norm{\vx}^2-\conj{\lambda}\inner{\vx}{\vy}-\lambda\inner{\vy}{\vx}+\abs{\lambda}^2\norm{\vy}^2=\norm{\vx}^2-\frac{\abs{\inner{\vx}{\vy}}^2}{\norm{\vy}^2},
\]
چون هر سه جمله‌ی آخر برابر $\frac{\abs{\inner{\vx}{\vy}}^2}{\norm{\vy}^2}$ هستند. تساوی یعنی $\vz=\zero$.
\end{proof}

\begin{note}
در فضای حقیقی این نامساوی اجازه می‌دهد زاویه را در هر فضایی تعریف کنیم: $\cos\theta=\frac{\inner{\vx}{\vy}}{\norm{\vx}\norm{\vy}}$ (تعمیم قضیه‌ی \ref{thm:dot-angle}). اگر $\inner{\vx}{\vy}=0$، دو بردار \term{متعامد}{Orthogonal} هستند.
\end{note}

\begin{theorem}\label{thm:inner-norm}
$\norm{\vx}=\sqrt{\inner{\vx}{\vx}}$ یک نُرم است.
\end{theorem}
\begin{proof}
مثبت معین بودن از شرط (ج) ضرب داخلی می‌آید. همگنی: $\inner{\alpha\vx}{\alpha\vx}=\alpha\conj{\alpha}\inner{\vx}{\vx}=\abs{\alpha}^2\norm{\vx}^2$. نامساوی مثلث با $\operatorname{Re}z\le\abs z$ و کوشی-شوارتز:
\[
\norm{\vx+\vy}^2=\norm{\vx}^2+\norm{\vy}^2+2\operatorname{Re}\inner{\vx}{\vy}\le\norm{\vx}^2+\norm{\vy}^2+2\norm{\vx}\norm{\vy}=(\norm{\vx}+\norm{\vy})^2.
\]
\end{proof}

\begin{note}[هر نُرمی از ضرب داخلی نمی‌آید]
نُرم‌های ناشی از ضرب داخلی \textbf{قانون متوازی‌الاضلاع} را دارند: $\norm{\vx+\vy}^2+\norm{\vx-\vy}^2=2\norm{\vx}^2+2\norm{\vy}^2$. برای $\norm{\vx}_\infty=\max\abs{x_i}$ با $\vx=(1,0)$ و $\vy=(0,1)$ سمت چپ ۲ و سمت راست ۴ است.
\end{note}

\begin{definition}[نُرم اقلیدسی]
\term{نُرم اقلیدسی}{Euclidean Norm} در $\R^n$: $\norm{\vx}_2=\sqrt{x_1^2+\dots+x_n^2}=\sqrt{\vx\T\vx}$ (همان طول، از فیثاغورس). برداری با طول ۱ \term{یکه}{Unit} است و $\frac{\vv}{\norm{\vv}}$ بردار یکه در جهت $\vv$ است.
\end{definition}

\begin{intuition}[«دایره‌ی واحد» در نُرم‌های مختلف]
\[
\norm{\vx}_1=\sum_i\abs{x_i},\qquad \norm{\vx}_2=\sqrt{\sum_ix_i^2},\qquad \norm{\vx}_\infty=\max_i\abs{x_i}.
\]

\begin{theorem}[نُرم مجموع]\label{thm:norm-sum}
در $\R^n$: $\norm{\vx+\vy}^2=\norm{\vx}^2+2\vx\T\vy+\norm{\vy}^2$. پس $\vx\perp\vy$ اگر و تنها اگر $\norm{\vx+\vy}^2=\norm{\vx}^2+\norm{\vy}^2$ (\textbf{فیثاغورس}).
\end{theorem}
\begin{proof}
$(\vx+\vy)\T(\vx+\vy)=\vx\T\vx+\vx\T\vy+\vy\T\vx+\vy\T\vy$ و $\vy\T\vx=\vx\T\vy$.
\end{proof}

\section{متر و فاصله}

\begin{definition}[متر]
تابع $d:V\times V\to\R$ یک \term{متر}{Metric} است اگر:
\begin{enumerate}[label=(\roman*)]
  \item $d(\vv,\vw)\ge0$ و $d(\vv,\vw)=0\iff\vv=\vw$؛
  \item $d(\vv,\vw)=d(\vw,\vv)$؛\footnote{در اسلایدها این شرط به اشتباه $d(v,w)=d(v,w)$ تایپ شده بود.}
  \item $d(\vv,\vw)\le d(\vv,\vu)+d(\vu,\vw)$.
\end{enumerate}
\textbf{فاصله‌ی اقلیدسی}: $\dist(\va,\vb)=\norm{\va-\vb}$.
\end{definition}

\begin{theorem}
اگر $\norm{\cdot}$ نُرم باشد، $d(\vx,\vy)=\norm{\vx-\vy}$ متر است.
\end{theorem}
\begin{proof}
\begin{enumerate}[label=(\roman*)]
  \item از مثبت معین بودن نُرم.
  \item $\norm{\vy-\vx}=\abs{-1}\,\norm{\vx-\vy}$.
  \item $\norm{\vx-\vz}=\norm{(\vx-\vy)+(\vy-\vz)}\le\norm{\vx-\vy}+\norm{\vy-\vz}$.
\end{enumerate}
\end{proof}

برعکس‌ها درست نیستند: نُرم بیشینه از ضرب داخلی نمی‌آید، و متر گسسته ($d=1$ برای $\vx\neq\vy$) از نُرم نمی‌آید، چون $d(2\vx,\zero)=1\neq2=2d(\vx,\zero)$.

\begin{example}[کدگذاری یک‌داغ]
فاصله‌ی هر دو کلمه‌ی متفاوت $\norm{\ve_i-\ve_j}=\sqrt2$ است؛ یعنی با معیار فاصله هم همه‌ی کلمه‌ها به یک اندازه از هم دورند (همان عیبی که در فصل ۱ دیدیم).
\end{example}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item هر تابع خطی $\R^n\to\R$ به شکل یکتای $\va\T\vx$ است؛ هر فرم دوخطی به شکل $\vv\T A\vw$ و $[f]_{\mathcal B}=P\T AP$.
  \item ضرب داخلی: (مزدوج‌)متقارن، خطی در آرگومان اول، مثبت معین. نمونه‌ها: $\vx\T\vy$، $\tr(X\T Y)$، $\int fg$.
  \item کوشی-شوارتز: $\abs{\inner{\vx}{\vy}}\le\norm\vx\norm\vy$.
  \item ضرب داخلی $\to$ نُرم $\to$ متر، اما برعکسشان لزوماً نه.
\end{itemize}
\end{summary}

\subsection*{تمرین‌ها}
\begin{enumerate}
  \item ماتریس $f(\vv,\vw)=v_1w_1-2v_1w_2+3v_2w_2$ را بنویسید. متقارن است؟
  \item آیا $\inner{\vx}{\vy}=x_1y_1-x_2y_2$ روی $\R^2$ ضرب داخلی است؟
  \item در $C[0,1]$ با $\inner{f}{g}=\int_0^1fg$، زاویه‌ی بین $1$ و $x$ را بیابید.
  \item نُرم‌های $\ell_1$، $\ell_2$ و $\ell_\infty$ بردار $(3,-4)$ را حساب کنید.
\end{enumerate}

\subsection*{پاسخ‌های کوتاه}
\begin{enumerate}
  \item $\begin{bmatrix}1&-2\\0&3\end{bmatrix}$؛ متقارن نیست.
  \item خیر؛ $\inner{(0,1)}{(0,1)}=-1<0$.
  \item $\inner{1}{x}=\frac12$، $\norm{1}=1$، $\norm{x}=\frac1{\sqrt3}$؛ پس $\cos\theta=\frac{\sqrt3}2$ و $\theta=30^\circ$.
  \item $7$، $5$ و $4$.
\end{enumerate}
