\chapter{زیرفضا}
\label{ch:subspaces}

درون یک فضای برداری، اغلب مجموعه‌های کوچک‌تری هستند که \emph{خودشان} هم فضای برداری‌اند؛ مثل خط یا صفحه‌ی گذرنده از مبدأ در $\R^3$. به این‌ها زیرفضا می‌گوییم. در این فصل می‌بینیم چطور زیرفضا بودن را بررسی کنیم و چطور از زیرفضاها، زیرفضای جدید بسازیم.

\section{تعریف و محک زیرفضا}

\begin{definition}[زیرفضا]
زیرمجموعه‌ی $U\subseteq V$ یک \term{زیرفضا}{Subspace} از $V$ است اگر $U$ با \textbf{همان} جمع و ضرب اسکالرِ $V$ خودش فضای برداری باشد. $\{\zero\}$ (\term{زیرفضای صفر}{Zero Subspace}) و خود $V$ همیشه زیرفضا هستند.
\end{definition}

لازم نیست هر هشت اصل را دوباره بررسی کنیم: اصولی مثل جابه‌جایی که برای \emph{همه‌ی} بردارهای $V$ درست‌اند، برای بردارهای $U$ هم درست‌اند. فقط باید مطمئن شویم با جمع و ضرب اسکالر از $U$ بیرون نمی‌رویم.

\begin{theorem}[محک زیرفضا]\label{thm:subspace-test}
زیرمجموعه‌ی $H\subseteq V$ زیرفضاست اگر و تنها اگر
\begin{enumerate}
  \item $\zero\in H$؛
  \item $\vu,\vv\in H\Rightarrow\vu+\vv\in H$؛
  \item $\vu\in H\Rightarrow c\vu\in H$.
\end{enumerate}
معادلاً: $H$ ناتهی باشد و برای هر $\vb,\vc\in H$ و هر اسکالر $\alpha$، $\alpha\vb+\vc\in H$.
\end{theorem}
\begin{proof}
\textbf{(رفت)} زیرفضا خودش فضای برداری است؛ پس صفر دارد (و $\zero_U=0\vx=\zero_V$ طبق قضیه‌ی \ref{thm:vs-basic}) و تحت هر دو عمل بسته است.

\textbf{(برگشت)} با شرط یک‌خطی: عضوی مثل $\vu_0\in H$ هست و $(-1)\vu_0+\vu_0=\zero\in H$. با $\vc=\zero$ بستاری ضرب اسکالر و با $\alpha=1$ بستاری جمع به دست می‌آید. قرینه هم در $H$ است: $-\vb=(-1)\vb$. بقیه‌ی اصول از $V$ به ارث می‌رسند.
\end{proof}

\begin{warning}
ساده‌ترین راه برای رد زیرفضا بودن: \textbf{آیا صفر در آن هست؟} اگر نباشد، قطعاً زیرفضا نیست (اما بودن صفر کافی نیست).
\end{warning}

\begin{example}
کدام زیرفضاست؟
(الف) خط $y=x+1$ در $\R^2$؛
(ب) مجموعه‌ی جواب‌های $A\vx=\zero$؛
(ج) $H=\{(s,t,0)\}$ در $\R^3$؛ آیا $\R^2$ زیرفضای $\R^3$ است؟
(د) ماتریس‌های متقارن در $M_{n,n}$.
\end{example}
\begin{solution}
\textbf{(الف) خیر}؛ $\zero$ روی خط نیست. فقط خط‌های \emph{گذرنده از مبدأ} $\{t\vd\}$ زیرفضا هستند.

\textbf{(ب) بله}؛ $A\zero=\zero$ و $A(\alpha\vu+\vv)=\alpha A\vu+A\vv=\zero$. اما جواب‌های $A\vx=\vb$ با $\vb\neq\zero$ زیرفضا نیستند، چون $\zero$ جواب نیست.

\textbf{(ج)} $H$ زیرفضاست (صفحه‌ی $xy$). اما $\R^2$ اصلاً زیرمجموعه‌ی $\R^3$ نیست: بردارهایش دو مؤلفه دارند.

\textbf{(د) بله}؛ $(\alpha A+B)\T=\alpha A\T+B\T=\alpha A+B$.
\end{solution}

\begin{note}
زیرفضاهای $\R^2$: $\{\zero\}$، خط‌های گذرنده از مبدأ و خود $\R^2$. در $\R^3$ صفحه‌های گذرنده از مبدأ هم اضافه می‌شوند. (اثبات این‌که زیرفضای دیگری نیست را با مفهوم بُعد در فصل ۶ می‌بینیم.)
\end{note}

\section{اشتراک و اجتماع}

\begin{theorem}\label{thm:intersection}
اشتراک هر تعداد زیرفضای $V$، زیرفضای $V$ است.
\end{theorem}
\begin{proof}
$\zero$ در همه‌ی زیرفضاهاست، پس در اشتراک است. اگر $\vb,\vc$ در اشتراک باشند، در هر کدام از زیرفضاها هستند؛ پس $\alpha\vb+\vc$ هم در هر کدام و در نتیجه در اشتراک است.
\end{proof}

\begin{theorem}[اجتماع زیرفضاها]\label{thm:union}
$W_1\cup W_2$ زیرفضاست اگر و تنها اگر $W_1\subseteq W_2$ یا $W_2\subseteq W_1$.
\end{theorem}
\begin{proof}
\textbf{(برگشت)} اگر $W_1\subseteq W_2$، اجتماع همان $W_2$ است.

\textbf{(رفت)} فرض کنید اجتماع زیرفضا باشد ولی $\vw_1\in W_1\setminus W_2$ و $\vw_2\in W_2\setminus W_1$ وجود داشته باشند. آن‌گاه $\vw_1+\vw_2$ در اجتماع است. اگر در $W_1$ باشد، $\vw_2=(\vw_1+\vw_2)-\vw_1\in W_1$؛ اگر در $W_2$ باشد، $\vw_1\in W_2$. هر دو تناقض است.
\end{proof}

\section{پوشش خطی (\lr{Span})}

\begin{definition}[پوشش خطی]
مجموعه‌ی همه‌ی ترکیب‌های خطی $\vv_1,\dots,\vv_p$ را \term{پوشش خطی}{Span} آن‌ها می‌نامیم:
\[
\Span\{\vv_1,\dots,\vv_p\}=\{c_1\vv_1+\dots+c_p\vv_p\ :\ c_i\in\F\}.
\]
اگر این مجموعه برابر $W$ باشد، می‌گوییم بردارها $W$ را \textbf{تولید می‌کنند} (\term{مجموعه‌ی مولد}{Spanning Set}). قرارداد: $\Span\emptyset=\{\zero\}$.
\end{definition}

\begin{intuition}
در $\R^3$، $\Span\{\vv\}$ خطی است که از مبدأ و نوک $\vv$ می‌گذرد، و اگر $\vu$ و $\vv$ هم‌راستا نباشند، $\Span\{\vu,\vv\}$ صفحه‌ای است که از مبدأ می‌گذرد و هر دو را دارد.
\end{intuition}

\begin{theorem}\label{thm:span-subspace}
$\Span\{\vv_1,\dots,\vv_p\}$ زیرفضای $V$ است و کوچک‌ترین زیرفضایی است که همه‌ی $\vv_i$ها را دارد.
\end{theorem}
\begin{proof}
$\zero=0\vv_1+\dots+0\vv_p$ در آن است و
$\alpha\sum b_i\vv_i+\sum c_i\vv_i=\sum(\alpha b_i+c_i)\vv_i$
دوباره ترکیب خطی است. اگر زیرفضای $W$ همه‌ی $\vv_i$ها را داشته باشد، چون بسته است، هر ترکیب خطی آن‌ها هم در $W$ است.
\end{proof}

\begin{note}
طبق قضیه‌ی \ref{thm:solvable-combination}، «آیا $\vb\in\Span\{\va_1,\dots,\va_n\}$؟» یعنی «آیا دستگاه $\aug{\va_1\ \cdots\ \va_n}{\vb}$ سازگار است؟».
\end{note}

\begin{example}
آیا $\vb=(7,4,3)\in\Span\{(1,-2,-5),(2,5,6)\}$؟
\end{example}
\begin{solution}
\[
\left[\begin{array}{cc|c}1&2&7\\-2&5&4\\-5&6&3\end{array}\right]
\sim
\left[\begin{array}{cc|c}1&2&7\\0&1&2\\0&0&6\end{array}\right].
\]
سطر آخر یعنی $0=6$؛ پس $\vb$ در این پوشش نیست.
\end{solution}

\section{جمع و جمع مستقیم}

\begin{definition}[جمع زیرفضاها]
$W_1+\dots+W_m=\{\vw_1+\dots+\vw_m: \vw_i\in W_i\}$.
\end{definition}

\begin{theorem}\label{thm:sum-subspace}
$W_1+\dots+W_m$ زیرفضای $V$ و کوچک‌ترین زیرفضای شامل همه‌ی $W_i$هاست.
\end{theorem}
\begin{proof}
$\zero=\zero+\dots+\zero$ در آن است و
$\alpha\sum\vw_i+\sum\vw_i'=\sum(\alpha\vw_i+\vw_i')$
که هر جمله‌اش در $W_i$ خودش است. هر زیرفضای شامل همه‌ی $W_i$ها، تحت جمع بسته است و همه‌ی این مجموع‌ها را دارد.
\end{proof}

پس جمع، جایگزین درستِ اجتماع است: $W_1+W_2=\Span(W_1\cup W_2)$. مثلاً جمع دو خط غیرموازی گذرنده از مبدأ در $\R^2$ کل صفحه است.

\begin{definition}[جمع مستقیم]
$U+W$ \term{جمع مستقیم}{Direct Sum} است و می‌نویسیم $U\oplus W$، اگر هر عضو آن \textbf{به‌طور یکتا} به شکل $\vu+\vw$ ($\vu\in U$، $\vw\in W$) نوشته شود.
\end{definition}

\begin{theorem}[محک جمع مستقیم]\label{thm:direct-sum}
$U+W$ جمع مستقیم است اگر و تنها اگر $U\cap W=\{\zero\}$.
\end{theorem}
\begin{proof}
\textbf{(رفت)} اگر $\vv\in U\cap W$، آن‌گاه $\vv=\vv+\zero=\zero+\vv$ دو نمایش است؛ پس $\vv=\zero$.

\textbf{(برگشت)} اگر $\vu+\vw=\vu'+\vw'$، آن‌گاه $\vu-\vu'=\vw'-\vw$ هم در $U$ است و هم در $W$؛ پس صفر است و $\vu=\vu'$، $\vw=\vw'$.
\end{proof}

\begin{example}
در $\R^3$: (الف) $U=\{(a,b,0)\}$ و $W=\{(0,0,c)\}$؛ (ب) $U=\{(a,b,0)\}$ و $W=\{(0,c,d)\}$.
\end{example}
\begin{solution}
در هر دو $U+W=\R^3$. در (الف) $U\cap W=\{\zero\}$؛ پس $\R^3=U\oplus W$. در (ب) محور $y$ مشترک است و نمایش یکتا نیست: $(0,1,0)=(0,1,0)+\zero=\zero+(0,1,0)$.
\end{solution}

\begin{warning}
برای سه زیرفضا یا بیشتر، صفر بودن اشتراک‌های دوبه‌دو کافی نیست: سه خط $x$، $y$ و $y=x$ در $\R^2$ دوبه‌دو فقط در $\zero$ مشترک‌اند، اما $\zero=(1,0)+(0,1)+(-1,-1)$.
\end{warning}

\begin{example}[زوج و فرد]
نشان دهید چندجمله‌ای‌ها جمع مستقیم چندجمله‌ای‌های زوج ($E$) و فرد ($O$) هستند.
\end{example}
\begin{solution}
هر $p$ را می‌شود نوشت
$p(t)=\frac{p(t)+p(-t)}{2}+\frac{p(t)-p(-t)}{2}$
که بخش اول زوج و بخش دوم فرد است؛ پس $P=E+O$. چندجمله‌ای‌ای که هم فقط توان زوج دارد و هم فقط توان فرد، صفر است؛ پس $E\cap O=\{0\}$ و $P=E\oplus O$.
\end{solution}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item محک زیرفضا: ناتهی و $\alpha\vb+\vc\in U$. اگر $\zero\notin U$، زیرفضا نیست.
  \item اشتراک زیرفضاها زیرفضاست؛ اجتماع فقط وقتی که یکی درون دیگری باشد.
  \item $\Span$ کوچک‌ترین زیرفضای شامل بردارهاست؛ $W_1+W_2$ کوچک‌ترین زیرفضای شامل هر دو.
  \item $U\oplus W\iff U\cap W=\{\zero\}$.
\end{itemize}
\end{summary}

\subsection*{تمرین‌ها}
\begin{enumerate}
  \item آیا $\{(x,y,z): x+y+z=0\}$ زیرفضای $\R^3$ است؟ $\{(x,y,z): x+y+z=1\}$ چطور؟
  \item آیا $\{(x,y)\in\R^2: xy\ge0\}$ زیرفضاست؟
  \item آیا $U=\{(a,a,0)\}$ و $W=\{(0,b,b)\}$ جمع مستقیم می‌سازند؟ $U\oplus W$ را با یک معادله توصیف کنید.
\end{enumerate}

\subsection*{پاسخ‌های کوتاه}
\begin{enumerate}
  \item اولی بله (جواب‌های یک دستگاه همگن)؛ دومی خیر ($\zero$ در آن نیست).
  \item خیر؛ $(1,0)+(0,-1)=(1,-1)$ در آن نیست.
  \item بله، $U\cap W=\{\zero\}$. $U\oplus W=\{(a,a+b,b)\}=\{(x,y,z): x-y+z=0\}$.
\end{enumerate}
