\chapter{دستگاه معادلات خطی، بردار و ماتریس}
\label{ch:linear-equations}

نقطه‌ی شروع درس یک سؤال ساده است: \emph{چند معادله‌ی خطی با چند مجهول داریم؛ جوابشان چیست؟} در این فصل بردار و ماتریس را معرفی می‌کنیم و می‌بینیم که حل دستگاه با چند «عمل سطری» ساده انجام می‌شود؛ عمل‌هایی که خودشان ضرب ماتریس‌ها هستند.

\section{دستگاه معادلات خطی}

\begin{definition}[دستگاه معادلات خطی]
فرض کنید $\F$ یک میدان باشد. یک \term{دستگاه $m$ معادله‌ی خطی با $n$ مجهول}{System of Linear Equations} به این شکل است:
\[
\begin{cases}
A_{11}x_1+A_{12}x_2+\dots+A_{1n}x_n=y_1\\
\qquad\vdots\\
A_{m1}x_1+A_{m2}x_2+\dots+A_{mn}x_n=y_m
\end{cases}
\]
که $A_{ij}$ و $y_k$ عضوهای $\F$ هستند ($A_{ij}$ ضریب $x_j$ در معادله‌ی $i$-ام است).
\begin{itemize}
  \item اگر $y_1=\dots=y_m=0$ باشد، دستگاه \term{همگن}{Homogeneous} است.
  \item \term{جواب}{Solution} دستگاه، برداری مثل $\vs=(s_1,\dots,s_n)$ است که با $x_1=s_1,\dots,x_n=s_n$ همه‌ی معادله‌ها درست شوند.
\end{itemize}
\end{definition}

\subsection{دو تعبیر هندسی}
دستگاه زیر را در نظر بگیرید:
\[
\begin{cases}
x-2y=1\\
3x+2y=11
\end{cases}
\]
\textbf{تصویر سطری:} هر معادله یک خط است و جواب، محل برخورد دو خط یعنی $(3,1)$ است.

\begin{center}
\begin{tikzpicture}[scale=0.62]
  \draw[step=1, gray!20] (-1,-2) grid (6,6);
  \draw[->] (-1,0)--(6.3,0) node[below] {$x$};
  \draw[->] (0,-2)--(0,6.3) node[left] {$y$};
  \draw[very thick, primary, domain=-1:6] plot (\x, {(\x-1)/2}) node[right] {$x-2y=1$};
  \draw[very thick, excolor, domain=1.5:4.4] plot (\x, {(11-3*\x)/2}) node[below right] {$3x+2y=11$};
  \fill[warncolor] (3,1) circle (4pt) node[above right=2pt] {$(3,1)$};
\end{tikzpicture}
\end{center}

\textbf{شکل ماتریسی:} همین دستگاه یعنی $A\vx=\vb$:
\[
\underbrace{\begin{bmatrix}1&-2\\3&2\end{bmatrix}}_{A}
\underbrace{\begin{bmatrix}x\\y\end{bmatrix}}_{\vx}
=
\underbrace{\begin{bmatrix}1\\11\end{bmatrix}}_{\vb},
\qquad
\aug{A}{\vb}=\left[\begin{array}{cc|c}1&-2&1\\3&2&11\end{array}\right].
\]
به $A$ \term{ماتریس ضرایب}{Coefficient Matrix} و به $\aug{A}{\vb}$ \term{ماتریس افزوده}{Augmented Matrix} می‌گوییم.

\textbf{تصویر ستونی:} همین دستگاه یعنی ترکیبی از ستون‌های $A$:
\[
x\begin{bmatrix}1\\3\end{bmatrix}+y\begin{bmatrix}-2\\2\end{bmatrix}=\begin{bmatrix}1\\11\end{bmatrix}.
\]
یعنی چه مقدار از ستون اول و چه مقدار از ستون دوم را جمع کنیم تا به $\vb$ برسیم؟ (۳ تا از اولی و ۱ تا از دومی.)

\subsection{ایده‌ی حذف}
مضربی از یک معادله را از دیگری کم می‌کنیم تا یک مجهول حذف شود:
\[
\begin{cases}
x-2y=1\\
3x+2y=11
\end{cases}
\quad\xrightarrow{\text{\fa{معادله‌ی دوم منهای ۳ برابر اولی}}}\quad
\begin{cases}
x-2y=1\\
\phantom{3x+{}}8y=8
\end{cases}
\]
حالا $y=1$ و با جایگذاری $x=3$. ماتریس ضرایب به یک ماتریس \term{بالامثلثی}{Upper Triangular} تبدیل شد (زیر قطرش صفر است). این ایده در فصل ۲ به الگوریتم \textbf{حذف گاوسی} تبدیل می‌شود.

\subsection{وجود و یکتایی جواب}

\begin{theorem}[تعداد جواب‌ها]\label{thm:num-solutions}
یک دستگاه خطی روی $\R$ یا $\C$ یا جواب ندارد، یا یک جواب، یا بی‌شمار جواب دارد.
\end{theorem}
\begin{proof}
اگر $\vu\neq\vv$ هر دو جواب $A\vx=\vb$ باشند، برای هر عدد $t$ بردار $\vu+t(\vv-\vu)$ هم جواب است، چون $A\vu+t(A\vv-A\vu)=\vb+t(\vb-\vb)=\vb$. این بردارها برای $t$های مختلف متفاوت‌اند؛ پس بی‌شمار جواب داریم.
\end{proof}

در کل درس دنبال دو سؤال هستیم: (۱) آیا دستگاه \textbf{سازگار} است؟ (۲) اگر جواب دارد، آیا \textbf{یکتاست}؟

\section{بردارها و ضرب داخلی}

یک \term{بردار}{Vector} در $\R^n$ فهرست مرتبی از $n$ عدد است که به‌صورت ستونی نوشته می‌شود؛ گاهی برای صرفه‌جویی می‌نویسیم $\vv=(v_1,\dots,v_n)$. جمع، تفریق و ضرب در اسکالر مؤلفه‌به‌مؤلفه انجام می‌شوند. از نظر هندسی جمع دو بردار همان قاعده‌ی متوازی‌الاضلاع است.

\subsection{داده‌های دسته‌ای و کدگذاری یک‌داغ}
در یادگیری ماشین برای داده‌های \term{دسته‌ای}{Categorical} (مثل رنگ یا کلمه) از \term{کدگذاری یک‌داغ}{One-Hot Encoding} استفاده می‌شود: به هر دسته برداری با یک مؤلفه‌ی ۱ و بقیه صفر می‌دهیم:
\[
\text{apple}=\begin{bmatrix}1\\0\\0\\0\end{bmatrix},\quad
\text{cat}=\begin{bmatrix}0\\1\\0\\0\end{bmatrix},\quad
\text{house}=\begin{bmatrix}0\\0\\1\\0\end{bmatrix},\quad
\text{tiger}=\begin{bmatrix}0\\0\\0\\1\end{bmatrix}.
\]
این‌ها همان بردارهای استاندارد $\ve_1,\dots,\ve_4$ هستند. دو عیب دارند: خیلی \textbf{تُنُک}‌اند (تقریباً همه‌ی مؤلفه‌ها صفر است) و هیچ دو کلمه‌ای \textbf{شبیه هم} نیستند.

\subsection{ضرب داخلی}
\begin{definition}[ضرب داخلی]
برای $\va,\vb\in\R^n$، \term{ضرب داخلی}{Inner/Dot Product} یک \textbf{عدد} است، نه بردار:
\[
\va\cdot\vb=\va\T\vb=a_1b_1+\dots+a_nb_n=\sum_{i=1}^n a_ib_i .
\]
نمادهای دیگر: $\inner{\va}{\vb}$ و $\langle\va|\vb\rangle$. مثلاً $(2,1)\cdot(1,5)=2+5=7$.
\end{definition}

\begin{theorem}[ویژگی‌های ضرب داخلی]\label{thm:dot-props}
برای هر $\va,\vb,\vc\in\R^n$ و عدد $\alpha$:
$\va\cdot\vb=\vb\cdot\va$، \ $\va\cdot(\vb+\vc)=\va\cdot\vb+\va\cdot\vc$، \ $(\alpha\va)\cdot\vb=\alpha(\va\cdot\vb)$، و $\va\cdot\va\ge0$ که فقط برای $\va=\zero$ صفر است.
\end{theorem}
همه مستقیم از تعریف به دست می‌آیند؛ آخری چون $\va\cdot\va=\sum a_i^2$. چون $\va\T\vb$ یک عدد است، با ترانهاده‌اش برابر است: $\va\T\vb=(\va\T\vb)\T=\vb\T\va$.

\textbf{طول بردار:} ضرب داخلی بردار در خودش، مربع طول آن است:
$\norm{\va}^2=\va\T\va=\sum a_i^2$. مثلاً برای $\vv=(4,3)$ داریم $\vv\cdot\vv=25$ و $\norm{\vv}=5$.

\begin{theorem}[ضرب داخلی و زاویه]\label{thm:dot-angle}
اگر $\theta$ زاویه‌ی بین دو بردار ناصفر $\va,\vb$ در $\R^2$ یا $\R^3$ باشد:
$\va\cdot\vb=\norm{\va}\,\norm{\vb}\cos\theta$.
\end{theorem}

\begin{intuition}[سایه]
اگر طول $\vb$ برابر ۱ باشد، $\va\cdot\vb=\norm{\va}\cos\theta$ طول \textbf{سایه‌ی} $\va$ روی امتداد $\vb$ است. پس ضرب داخلی نشان می‌دهد دو بردار چقدر هم‌جهت یا \textbf{شبیه} هم‌اند؛ اگر صفر باشد، بردارها بر هم \term{عمود}{Orthogonal} هستند.

  \draw[->, very thick, excolor] (0,0)--(3,2.2) node[above] {$\va$};
  \draw[dashed] (3,2.2)--(3,0);

\begin{example}[عیب کدگذاری یک‌داغ]
آیا «گربه» به «ببر» شبیه‌تر است تا به «سیب»؟
\end{example}
\begin{solution}
خیر: $\text{apple}\cdot\text{cat}=0=\text{tiger}\cdot\text{cat}$. در کل $\ve_i\cdot\ve_j=0$ برای $i\neq j$؛ همه‌ی کلمه‌ها بر هم عمودند و هیچ شباهتی دیده نمی‌شود.
\end{solution}

\section{ماتریس‌ها و ضرب آن‌ها}\label{sec:matmul}

منظور از $A\in\R^{m\times n}$ ماتریسی با $m$ سطر و $n$ ستون است. آن را می‌توان ستون‌به‌ستون یا سطربه‌سطر دید:
\[
A=\begin{bmatrix}
a_{11}&\cdots&a_{1n}\\
\vdots&\ddots&\vdots\\
a_{m1}&\cdots&a_{mn}
\end{bmatrix}
=
\begin{bmatrix}
| & & |\\
\va_1 & \cdots & \va_n\\
| & & |
\end{bmatrix}
=
\begin{bmatrix}
- & \va_1\T & -\\
 & \vdots & \\
- & \va_m\T & -
\end{bmatrix}.
\]
ماتریس \textbf{همانی} $I$ روی قطر ۱ و جای دیگر ۰ است ($I_{ij}=\delta_{ij}$). \term{ترانهاده}{Transpose} با $(A\T)_{ij}=A_{ji}$ تعریف می‌شود.

\begin{definition}[ترکیب خطی]
یک \term{ترکیب خطی}{Linear Combination} از بردارهای $\va_1,\dots,\va_m$ هر برداری به شکل $\beta_1\va_1+\dots+\beta_m\va_m$ است که $\beta_i$ها اسکالرند.
\end{definition}

\subsection{ضرب ماتریس در بردار}
برای $A\in\R^{m\times n}$ و $\vx\in\R^n$، حاصل $A\vx\in\R^m$ با $(A\vx)_i=\sum_j a_{ij}x_j$ تعریف می‌شود. دو نگاه به آن:
\[
\text{\fa{سطری:}}\ \ A\vx=\begin{bmatrix}\va_1\T\vx\\ \vdots\\ \va_m\T\vx\end{bmatrix},
\qquad
\text{\fa{ستونی:}}\ \ A\vx=x_1\va_1+x_2\va_2+\dots+x_n\va_n .
\]
\begin{note}
\hl{$A\vx$ همیشه ترکیب خطی ستون‌های $A$ است} و به همین ترتیب $\vx\T A$ ترکیب خطی سطرهای $A$ است. بعداً می‌بینیم ستون‌های $A$ مستقل خطی‌اند اگر و تنها اگر از $A\vx=\zero$ نتیجه شود $\vx=\zero$.
\end{note}
ویژگی‌ها: $A(\vu+\vv)=A\vu+A\vv$، \ $(A+B)\vu=A\vu+B\vu$، \ $(\alpha A)\vu=\alpha(A\vu)=A(\alpha\vu)$، \ $A\zero=\zero$ و $I\vu=\vu$.

\begin{example}
برای $A=\begin{bmatrix}-1&2&1\\2&0&-2\end{bmatrix}$، ستون $j$-ام، سطر $i$-ام، مجموع سطرها و مجموع ستون‌ها را به شکل ضرب ماتریس و بردار بنویسید.
\end{example}
\begin{solution}
ستون $j$-ام: $A\ve_j$. سطر $i$-ام: $\ve_i\T A$. مجموع سطرها: $\begin{bmatrix}1&1\end{bmatrix}A=\begin{bmatrix}1&2&-1\end{bmatrix}$. مجموع ستون‌ها: $A\begin{bmatrix}1\\1\\1\end{bmatrix}=\begin{bmatrix}2\\0\end{bmatrix}$.
\end{solution}

\subsection{ضرب ماتریس در ماتریس}
\begin{definition}[ضرب دو ماتریس]
اگر $A$ از اندازه‌ی $m\times n$ و $B$ از اندازه‌ی $n\times p$ باشد، $C=AB$ ماتریسی $m\times p$ است با
\[
C_{ij}=\sum_{r=1}^{n}A_{ir}B_{rj}=\va_i\T\vb_j ,
\]
یعنی ضرب داخلی سطر $i$-ام $A$ در ستون $j$-ام $B$. ضرب فقط وقتی تعریف شده که تعداد ستون‌های $A$ با تعداد سطرهای $B$ برابر باشد.
\end{definition}

\textbf{چهار نگاه به $AB$:}
\begin{enumerate}
  \item \textbf{ضرب‌های داخلی:} درایه‌ی $(i,j)$ برابر $\va_i\T\vb_j$ است.
  \item \textbf{مجموع ضرب‌های خارجی:} $AB=\sum_{r=1}^{n}\va_r\vb_r\T$ (ستون $r$-ام $A$ ضرب در سطر $r$-ام $B$).
  \item \textbf{ستون‌به‌ستون:} $AB=\begin{bmatrix}A\vb_1&\cdots&A\vb_p\end{bmatrix}$؛ پس هر ستون $AB$ ترکیب خطی ستون‌های $A$ است.
  \item \textbf{سطربه‌سطر:} سطر $i$-ام $AB$ برابر $\va_i\T B$ است؛ پس \hl{هر سطر $AB$ ترکیب خطی سطرهای $B$ است.}
\end{enumerate}

\textbf{ویژگی‌ها:} ضرب ماتریس‌ها \textbf{شرکت‌پذیر} است، $(AB)C=A(BC)$، و \textbf{توزیع‌پذیر}، $A(B+C)=AB+AC$؛ همچنین $IA=AI=A$.

\begin{warning}[ضرب ماتریس‌ها جابه‌جایی نیست]
در حالت کلی $AB\neq BA$؛ گاهی حتی یکی از دو ضرب تعریف نشده است. مثلاً
\[
A=\begin{bmatrix}0&1\\0&0\end{bmatrix},\ B=\begin{bmatrix}0&0\\1&0\end{bmatrix}:\qquad
AB=\begin{bmatrix}1&0\\0&0\end{bmatrix}\neq\begin{bmatrix}0&0\\0&1\end{bmatrix}=BA .
\]
\end{warning}

\begin{theorem}[ترانهاده و وارون]\label{thm:transpose}\label{thm:inverse-props}
\begin{enumerate}
  \item $(AB)\T=B\T A\T$.
  \item ماتریس مربعی $A$ \term{وارون‌پذیر}{Invertible} است اگر ماتریسی مثل $A^{-1}$ باشد که $AA^{-1}=A^{-1}A=I$. وارون یکتاست و اگر $A,B$ وارون‌پذیر باشند، $(AB)^{-1}=B^{-1}A^{-1}$.
\end{enumerate}
\end{theorem}
برای بند ۲: $(AB)(B^{-1}A^{-1})=A(BB^{-1})A^{-1}=I$. برای ماتریس $2\times2$ اگر $ad-bc\neq0$: $\begin{bmatrix}a&b\\c&d\end{bmatrix}^{-1}=\frac{1}{ad-bc}\begin{bmatrix}d&-b\\-c&a\end{bmatrix}$.

\section{عملیات سطری مقدماتی}

\begin{definition}[عملیات سطری مقدماتی]
سه نوع \term{عمل سطری مقدماتی}{Elementary Row Operation} داریم:
\begin{enumerate}
  \item \textbf{مقیاس‌دهی:} ضرب یک سطر در عدد \emph{ناصفر} $c$: \ $R_i\leftarrow cR_i$.
  \item \textbf{جایگزینی:} جمع یک سطر با مضربی از سطر دیگر: \ $R_i\leftarrow R_i+cR_k$.
  \item \textbf{جابه‌جایی:} عوض کردن جای دو سطر: \ $R_i\leftrightarrow R_k$.
\end{enumerate}
هر عمل سطری یک تابع $e$ است که به هر ماتریس $m\times n$ مثل $A$ یک ماتریس $m\times n$ مثل $e(A)$ نسبت می‌دهد؛ مثلاً برای جایگزینی سطر $r$: $e(A)_{rj}=A_{rj}+cA_{kj}$ و بقیه‌ی سطرها بدون تغییر می‌مانند. تعداد ستون‌های $A$ مهم نیست، اما تعداد سطرهایش مهم است.
\end{definition}

\begin{theorem}[وارون عمل سطری]\label{thm:inverse-rowop}
هر عمل سطری مقدماتی $e$ یک وارون $e_1$ از \textbf{همان نوع} دارد: $e_1(e(A))=e(e_1(A))=A$.
\end{theorem}
\begin{proof}
\begin{itemize}
  \item ضرب سطر $r$ در $c\neq0$ با ضرب همان سطر در $c^{-1}$ خنثی می‌شود (این‌جا به $c\neq0$ نیاز داریم).
  \item جایگزینی $R_r\leftarrow R_r+cR_s$ با $R_r\leftarrow R_r-cR_s$ خنثی می‌شود، چون سطر $s$ تغییر نکرده است.
  \item جابه‌جایی دو سطر، وارون خودش است.
\end{itemize}
\end{proof}

\begin{definition}[هم‌ارز سطری]
ماتریس $B$ با $A$ \term{هم‌ارز سطری}{Row-Equivalent} است اگر با تعداد متناهی عمل سطری مقدماتی از $A$ به دست آید.
\end{definition}

\begin{note}
از قضیه‌ی \ref{thm:inverse-rowop} نتیجه می‌شود: هر ماتریس با خودش هم‌ارز سطری است؛ اگر $B$ با $A$ هم‌ارز باشد، $A$ هم با $B$ هم‌ارز است (عمل‌های وارون را به ترتیب برعکس اعمال کنید)؛ و اگر $C$ با $B$ و $B$ با $A$ هم‌ارز باشد، $C$ با $A$ هم‌ارز است.
\end{note}

\begin{theorem}\label{thm:row-equiv-solutions}
اگر ماتریس‌های افزوده‌ی دو دستگاه هم‌ارز سطری باشند، دو دستگاه جواب‌های یکسانی دارند.
\end{theorem}
دلیلش این است که هر عمل سطری جواب‌های دستگاه قبلی را حفظ می‌کند و چون وارون‌پذیر است، برعکسش هم درست است.

\section{ماتریس‌های مقدماتی}

\begin{definition}[ماتریس مقدماتی]
ماتریس $m\times m$ را \term{مقدماتی}{Elementary Matrix} می‌نامیم اگر با \textbf{دقیقاً یک} عمل سطری مقدماتی از $I_m$ به دست آید.
\end{definition}

\begin{example}
همه‌ی ماتریس‌های مقدماتی $2\times2$ را بنویسید.
\end{example}
\begin{solution}
مقیاس‌دهی ($c\neq0$): $\begin{bmatrix}c&0\\0&1\end{bmatrix}$ و $\begin{bmatrix}1&0\\0&c\end{bmatrix}$؛ \
جایگزینی: $\begin{bmatrix}1&c\\0&1\end{bmatrix}$ و $\begin{bmatrix}1&0\\c&1\end{bmatrix}$؛ \
جابه‌جایی: $\begin{bmatrix}0&1\\1&0\end{bmatrix}$.
\end{solution}

\begin{theorem}\label{thm:elem-matrix}
اگر $e$ یک عمل سطری مقدماتی و $E=e(I_m)$ باشد، برای هر ماتریس $m\times n$ مثل $A$:
\[
e(A)=EA .
\]
\end{theorem}
\begin{proof}
$(EA)_{ij}=\sum_k E_{ik}A_{kj}$. برای سطرهایی که عمل به آن‌ها دست نمی‌زند، سطر $i$-ام $E$ همان سطر $I$ است ($E_{ik}=\delta_{ik}$) و $(EA)_{ij}=A_{ij}$. برای سطر $r$ که عمل رویش انجام می‌شود:
\begin{itemize}
  \item مقیاس‌دهی: $E_{rk}=c\,\delta_{rk}$، پس $(EA)_{rj}=c\,A_{rj}$.
  \item جایگزینی: $E_{rk}=\delta_{rk}+c\,\delta_{sk}$، پس $(EA)_{rj}=A_{rj}+c\,A_{sj}$.
  \item جابه‌جایی: $E_{rk}=\delta_{sk}$ و $E_{sk}=\delta_{rk}$، پس سطرهای $r$ و $s$ جابه‌جا می‌شوند.
\end{itemize}
در هر سه حالت $EA=e(A)$.
\end{proof}

\begin{intuition}
\hl{ضرب از چپ در یک ماتریس مربعی، عملیات سطری انجام می‌دهد}؛ چون هر سطر $EA$ ترکیب خطی سطرهای $A$ است.
\end{intuition}

\begin{example}
ماتریس $A=\begin{bmatrix}1&0&2\\-2&0&-3\\0&2&0\end{bmatrix}$ را با عملیات سطری به $I$ برسانید.
\end{example}
\begin{solution}
\[
A\xrightarrow{R_2\leftarrow R_2+2R_1}\begin{bmatrix}1&0&2\\0&0&1\\0&2&0\end{bmatrix}
\xrightarrow{R_2\leftrightarrow R_3}\begin{bmatrix}1&0&2\\0&2&0\\0&0&1\end{bmatrix}
\]
\[
\xrightarrow{R_2\leftarrow\frac12R_2}\begin{bmatrix}1&0&2\\0&1&0\\0&0&1\end{bmatrix}
\xrightarrow{R_1\leftarrow R_1-2R_3}I .
\]
اگر ماتریس‌های مقدماتی این چهار عمل $E_1,\dots,E_4$ باشند، طبق قضیه‌ی \ref{thm:elem-matrix} و شرکت‌پذیری:
$(E_4E_3E_2E_1)A=I$.
\end{solution}

\begin{theorem}\label{thm:row-equiv-PA}
ماتریس $B$ با $A$ هم‌ارز سطری است اگر و تنها اگر $B=PA$ که $P$ حاصل‌ضرب چند ماتریس مقدماتی $m\times m$ است.
\end{theorem}
\begin{proof}
اگر $B=e_k(\cdots e_1(A)\cdots)$ و $E_i=e_i(I)$، طبق قضیه‌ی \ref{thm:elem-matrix} و شرکت‌پذیری $B=E_k(\cdots(E_1A))=(E_k\cdots E_1)A$. برعکس، اگر $B=E_k\cdots E_1A$، هر ضرب در $E_i$ همان عمل سطری $e_i$ است؛ پس $B$ با این عمل‌ها از $A$ به دست می‌آید.
\end{proof}

\begin{corollary}\label{cor:elem-invertible}
هر ماتریس مقدماتی وارون‌پذیر است و وارونش ماتریس مقدماتیِ عمل وارون است؛ پس $P$ در قضیه‌ی بالا وارون‌پذیر است.
\end{corollary}
دلیل: اگر $E=e(I)$ و $E_1=e_1(I)$، آن‌گاه $E_1E=e_1(e(I))=I$ و $EE_1=I$.

\begin{example}[دستگاه همگن روی $\C$]
اگر ماتریس ضرایب یک دستگاه همگن روی اعداد مختلط
$A=\begin{bmatrix}-1&i\\-i&3\\1&2\end{bmatrix}$
باشد، همه‌ی جواب‌ها را بیابید.
\end{example}
\begin{solution}
با $i^2=-1$:
\[
A\xrightarrow[R_3\leftarrow R_3+R_1]{R_2\leftarrow R_2-iR_1}
\begin{bmatrix}-1&i\\0&4\\0&2+i\end{bmatrix}
\xrightarrow{R_2\leftarrow\frac14R_2}
\begin{bmatrix}-1&i\\0&1\\0&2+i\end{bmatrix}
\xrightarrow[R_3\leftarrow R_3-(2+i)R_2]{R_1\leftarrow R_1-iR_2}
\begin{bmatrix}-1&0\\0&1\\0&0\end{bmatrix}.
\]
پس $x_1=x_2=0$: فقط جواب بدیهی $\vx=\zero$.
\end{solution}

\section{جمع‌بندی و تمرین}

\begin{summary}
\begin{itemize}
  \item هر دستگاه خطی به شکل $A\vx=\vb$ است و یا جواب ندارد، یا یک جواب، یا بی‌شمار جواب.
  \item $\va\cdot\vb=\va\T\vb$ یک عدد است و $\norm{\va}^2=\va\T\va$.
  \item $A\vx$ ترکیب خطی ستون‌های $A$ است. ضرب ماتریس‌ها شرکت‌پذیر است اما جابه‌جایی نیست.
  \item هر عمل سطری مقدماتی وارون‌پذیر است و برابر ضرب از چپ در یک ماتریس مقدماتی است: $e(A)=EA$. پس $A\sim B\iff B=PA$.
\end{itemize}
\end{summary}

\subsection*{تمرین‌ها}
\begin{enumerate}
  \item برای $A=\begin{bmatrix}1&2\\0&1\end{bmatrix}$ و $B=\begin{bmatrix}1&0\\3&1\end{bmatrix}$، حاصل $AB$ و $BA$ را حساب کنید.
  \item نشان دهید از $AB=0$ لزوماً $A=0$ یا $B=0$ نتیجه نمی‌شود.
  \item زاویه‌ی بین $(1,1,0)$ و $(1,0,1)$ را پیدا کنید.
  \item ماتریس مقدماتی $3\times3$ عمل $R_3\leftarrow R_3-4R_1$ و وارونش را بنویسید.
\end{enumerate}

\subsection*{پاسخ‌های کوتاه}
\begin{enumerate}
  \item $AB=\begin{bmatrix}7&2\\3&1\end{bmatrix}$ و $BA=\begin{bmatrix}1&2\\3&7\end{bmatrix}$.
  \item $A=B=\begin{bmatrix}0&1\\0&0\end{bmatrix}$.
  \item $\cos\theta=\frac{1}{\sqrt2\sqrt2}=\frac12$، پس $\theta=60^\circ$.
  \item $E$ همان $I$ با $-4$ در جای $(3,1)$ است و $E^{-1}$ همان $I$ با $+4$ در همان جا.
\end{enumerate}
